local Maid = {}
Maid.__index = Maid

function Maid.new()
	return setmetatable({ _items = {}, _dead = false }, Maid)
end

function Maid:give(v)
	if self._dead then
		pcall(function()
			if typeof(v) == "RBXScriptConnection" then v:Disconnect()
			elseif typeof(v) == "Instance" then v:Destroy()
			elseif typeof(v) == "thread" then task.cancel(v)
			elseif type(v) == "function" then v() end
		end)
		return v
	end
	table.insert(self._items, v)
	return v
end

function Maid:clean()
	local items = self._items
	self._items = {}
	for i = #items, 1, -1 do
		local v = items[i]
		local ok, err = pcall(function()
			if typeof(v) == "RBXScriptConnection" then
				v:Disconnect()
			elseif typeof(v) == "Instance" then
				v:Destroy()
			elseif typeof(v) == "thread" then
				if coroutine.status(v) ~= "dead" then task.cancel(v) end
			elseif type(v) == "function" then
				v()
			elseif type(v) == "table" and v.clean then
				v:clean()
			end
		end)
		if not ok then
			local _ = err
		end
	end
end

function Maid:destroy()
	self:clean()
	self._dead = true
end

local Claims = setmetatable({}, { __mode = "k" })

local function claimSet(holder, obj, prop, value)
	local props = Claims[obj]
	if not props then
		props = {}
		Claims[obj] = props
	end
	local claim = props[prop]
	if not claim then
		local ok, old = pcall(function() return obj[prop] end)
		if not ok then return end
		claim = { original = { old }, holders = {}, count = 0 }
		props[prop] = claim
	end
	if not claim.holders[holder] then
		claim.holders[holder] = true
		claim.count += 1
	end
	pcall(function() obj[prop] = value end)
end

local function Restorer()
	local token = {}
	local held = {}
	local seen = {}
	local r = {}

	local function track(obj, prop)
		local s = seen[obj]
		if not s then
			s = {}
			seen[obj] = s
		end
		if not s[prop] then
			s[prop] = true
			table.insert(held, { obj = obj, prop = prop })
		end
	end

	function r.set(obj, prop, value)
		if typeof(obj) ~= "Instance" then return end
		track(obj, prop)
		claimSet(token, obj, prop, value)
	end

	function r.setFrom(obj, prop, value, original)
		if typeof(obj) ~= "Instance" then return end
		track(obj, prop)
		local props = Claims[obj]
		if not props then
			props = {}
			Claims[obj] = props
		end
		if not props[prop] then
			props[prop] = { original = { original }, holders = {}, count = 0 }
		end
		claimSet(token, obj, prop, value)
	end

	function r.restoreAll()
		for i = #held, 1, -1 do
			local e = held[i]
			held[i] = nil
			local props = Claims[e.obj]
			local claim = props and props[e.prop]
			if claim and claim.holders[token] then
				claim.holders[token] = nil
				claim.count -= 1
				if claim.count <= 0 then
					pcall(function() e.obj[e.prop] = claim.original[1] end)
					props[e.prop] = nil
				end
			end
		end
		seen = {}
	end

	return r
end

local Scheduler = {}
do
	local buckets = {
		render    = { signal = RunService.RenderStepped, list = {}, conn = nil, steppedArgs = false },
		stepped   = { signal = RunService.Stepped,       list = {}, conn = nil, steppedArgs = true },
		heartbeat = { signal = RunService.Heartbeat,     list = {}, conn = nil, steppedArgs = false },
	}

	Scheduler.cost = {}
	Scheduler.profiling = true

	local function pump(bucket)
		return function(a, b)
			local dt = bucket.steppedArgs and b or a
			local list = bucket.list
			local profiling = Scheduler.profiling
			for i = #list, 1, -1 do
				local entry = list[i]
				local ok, err
				if profiling then
					local t0 = os.clock()
					ok, err = pcall(entry.fn, dt)
					Scheduler.cost[entry.key] = (os.clock() - t0) * 1000
				else
					ok, err = pcall(entry.fn, dt)
				end
				if not ok then
					table.remove(list, i)
					Scheduler.cost[entry.key] = nil
					if entry.onError then
						task.spawn(entry.onError, err)
					else
						local _ = err
					end
				end
			end
		end
	end

	function Scheduler.add(name, key, fn, onError)
		local bucket = buckets[name]
		if not bucket then return end
		Scheduler.remove(name, key)
		table.insert(bucket.list, { key = key, fn = fn, onError = onError })
		if not bucket.conn then
			bucket.conn = bucket.signal:Connect(pump(bucket))
		end
	end

	function Scheduler.remove(name, key)
		local bucket = buckets[name]
		if not bucket then return end
		for i = #bucket.list, 1, -1 do
			if bucket.list[i].key == key then table.remove(bucket.list, i) end
		end
		Scheduler.cost[key] = nil
		if #bucket.list == 0 and bucket.conn then
			bucket.conn:Disconnect()
			bucket.conn = nil
		end
	end

	function Scheduler.stopAll()
		for _, bucket in pairs(buckets) do
			if bucket.conn then bucket.conn:Disconnect() end
			bucket.conn, bucket.list = nil, {}
		end
		Scheduler.cost = {}
	end
end

local collideBaseline = setmetatable({}, { __mode = "k" })

local function baseCollide(part)
	local v = collideBaseline[part]
	if v == nil then
		v = part.CanCollide
		collideBaseline[part] = v
	end
	return v
end

local function charParts()
	if not partCacheValid then
		partCache = {}
		if Char.model then
			for _, d in ipairs(Char.model:GetDescendants()) do
				if d:IsA("BasePart") then table.insert(partCache, d) end
			end
		end
		partCacheValid = true
	end
	return partCache
end

local function bindCharacter(model)
	charMaid:clean()
	invalidateParts()
	Char.model = model
	Char.hum   = model:FindFirstChildOfClass("Humanoid")
	Char.root  = model:FindFirstChild("HumanoidRootPart")

	task.spawn(function()
		if not Char.hum then Char.hum = model:WaitForChild("Humanoid", 10) end
		if not Char.root then Char.root = model:WaitForChild("HumanoidRootPart", 10) end
		invalidateParts()

		for _, p in ipairs(charParts()) do baseCollide(p) end
		for _, fn in ipairs(Char.onRespawn) do
			task.spawn(fn)
		end
	end)

	charMaid:give(model.DescendantAdded:Connect(invalidateParts))
	charMaid:give(model.DescendantRemoving:Connect(invalidateParts))
end

if LP.Character then bindCharacter(LP.Character) end
rootMaid:give(LP.CharacterAdded:Connect(bindCharacter))

local function alive()
	return (Char.model ~= nil and Char.model.Parent ~= nil
		and Char.hum ~= nil and Char.hum.Health > 0
		and Char.root ~= nil and Char.root.Parent ~= nil)
end

local function awaitAlive(timeout)
	local deadline = os.clock() + (timeout or 5)
	while not alive() and os.clock() < deadline do
		task.wait(0.1)
	end
	return alive()
end

local function flatLook(v)
	local flat = Vector3.new(v.X, 0, v.Z)
	if flat.Magnitude < 1e-4 then return Vector3.new(0, 0, -1) end
	return flat.Unit
end

local function safeLookAt(position, direction)
	return CFrame.lookAt(position, position + flatLook(direction))
end

local function isFinite(v)
	return v == v and v ~= math.huge and v ~= -math.huge
end

local function safeVector(v)
	if not (isFinite(v.X) and isFinite(v.Y) and isFinite(v.Z)) then
		return Vector3.zero
	end
	return v
end

	spawned[o] = true

	for k, v in pairs(props or {}) do
		if k == "Parent" then
			parent = v
		elseif k == "Name" then

		elseif k == "Font" then
			if typeof(v) == "Font" then o.FontFace = v else o.Font = v end

			o[k] = v

	btn.MouseEnter:Connect(function()
		tw(btn, MOTION.hover, { BackgroundColor3 = hoverColor })

local function buildFade(root)
	local elems = {}

	local function add(inst, prop)
		local ok, base = pcall(function() return inst[prop] end)
		if ok and type(base) == "number" and base < 1 then
			table.insert(elems, { inst = inst, prop = prop, base = base })
		end
	end

	add(root, "BackgroundTransparency")
	for _, d in ipairs(root:GetDescendants()) do
		if d:IsA("UIStroke") then
			add(d, "Transparency")
		elseif d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			add(d, "TextTransparency")
			add(d, "BackgroundTransparency")
		elseif d:IsA("ImageLabel") or d:IsA("ImageButton") then
			add(d, "ImageTransparency")
			add(d, "BackgroundTransparency")
		elseif d:IsA("GuiObject") then
			add(d, "BackgroundTransparency")
		end
	end
	return elems
end

local function setFade(elems, alpha)
	for _, e in ipairs(elems) do
		e.inst[e.prop] = e.base + (1 - e.base) * alpha
	end
end

local function tweenFade(elems, alpha, info, delay)
	local tweenInfo = info or MOTION.enter
	if delay and delay > 0 then
		tweenInfo = TweenInfo.new(
			tweenInfo.Time, tweenInfo.EasingStyle, tweenInfo.EasingDirection,
			0, false, delay
		)
	end
	for _, e in ipairs(elems) do
		TweenService:Create(e.inst, tweenInfo, {
			[e.prop] = e.base + (1 - e.base) * alpha,
		}):Play()
	end
end

local logLines = {}

local function pushLog(text)
	table.insert(logLines, ("[%s]  %s"):format(os.date("%H:%M:%S"), text))
	if #logLines > 400 then table.remove(logLines, 1) end
	if logRefresh then logRefresh() end
end

local notifySeq = 0
local liveNotifications = 0
local MAX_NOTIFICATIONS = 5

local function notify(text, kind, seconds)
	pushLog(text)

	if liveNotifications >= MAX_NOTIFICATIONS then
		local oldest, oldestOrder = nil, math.huge
		for _, c in ipairs(notifyRoot:GetChildren()) do
			if c:IsA("Frame") and c.LayoutOrder < oldestOrder then
				oldest, oldestOrder = c, c.LayoutOrder
			end
		end
		if oldest then
			liveNotifications -= 1
			oldest:Destroy()
		end
	end

	notifySeq += 1
	liveNotifications += 1

	local accent = (kind == "bad" and T.red) or (kind == "warn" and T.yellow) or T.green

	local card = new("Frame", {
		BackgroundColor3 = T.panel,
		Size = UDim2.new(1, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		LayoutOrder = notifySeq,
		ZIndex = 900,
		Parent = notifyRoot,
	})
	corner(card, 12)
	stroke(card, T.panelLine)
	pad(card, 12, 12, 13, 13)
	hlist(card, 11, Enum.VerticalAlignment.Top)

	new("Frame", {
		BackgroundColor3 = accent,
		Size = offset(3, 18),
		BorderSizePixel = 0,
		LayoutOrder = 1,
		ZIndex = 901,
		Parent = card,
	}, { new("UICorner", { CornerRadius = UDim.new(1, 0) }) })

	new("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.new(1, -14, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		Font = FONT,
		Text = text,
		TextSize = 15,
		TextColor3 = T.text,
		TextWrapped = true,
		TextXAlignment = Enum.TextXAlignment.Left,
		LayoutOrder = 2,
		ZIndex = 901,
		Parent = card,
	})

	local fade = buildFade(card)
	setFade(fade, 1)
	card.Position = offset(26, 0)
	tweenFade(fade, 0, MOTION.enter)
	tw(card, MOTION.enter, { Position = offset(0, 0) })

	task.delay(seconds or 3.5, function()
		if not card.Parent then return end
		liveNotifications -= 1
		tweenFade(fade, 1, MOTION.exit)
		local out = tw(card, MOTION.exit, { Position = offset(26, 0) })
		out.Completed:Wait()
		card:Destroy()
	end)
end

local lastNotifyAt = {}
local function notifyThrottled(id, gap, text, kind)
	local now = os.clock()
	if lastNotifyAt[id] and now - lastNotifyAt[id] < gap then return end
	lastNotifyAt[id] = now
	notify(text, kind)
end

	end
	local ok, err = pcall(function()
		if not isfolder(CFG_FOLDER) then makefolder(CFG_FOLDER) end
		writefile(cfgFile(), HttpService:JSONEncode(tbl))
	end)
	return ok, err
end

function Config.load()
	if not (Env.readfile and Env.isfile) then return nil, "this executor has no file API" end
	local ok, data = pcall(function()
		if not isfile(cfgFile()) then return nil end
		return HttpService:JSONDecode(readfile(cfgFile()))
	end)
	if not ok then return nil, tostring(data) end
	return data
end

local function doMinimize()
	if minimized or animatingMinimize then return end
	animatingMinimize = true
	minimized = true
	if closeActivePopup then closeActivePopup() end

	local pos = Vector2.new(shell.Position.X.Offset, shell.Position.Y.Offset)
	local size = Vector2.new(shell.AbsoluteSize.X, shell.AbsoluteSize.Y)
	if not fullscreen then shellRect.size, shellRect.pos = size, pos end

	local vp = viewport()
	local target = puckHome or Vector2.new(
		px(pos.X + size.X / 2 - PUCK / 2),
		px(pos.Y + size.Y / 2 - PUCK / 2)
	)
	target = Vector2.new(
		math.clamp(target.X, 0, math.max(0, px(vp.X - PUCK))),
		math.clamp(target.Y, 0, math.max(0, px(vp.Y - PUCK)))
	)
	puckHome = target

	shellFadeTo(1, MIN_FADE)
	chromeFadeTo(1, MIN_FADE)
	local drift = tw(shell, TweenInfo.new(MIN_MOVE, Enum.EasingStyle.Quint, Enum.EasingDirection.In), {
		Size = offset(size.X * 0.94, size.Y * 0.94),
		Position = offset(pos.X + size.X * 0.03, pos.Y + size.Y * 0.03),
	})

	task.delay(MIN_FADE, function()
		drift:Cancel()
		shell.Visible = false
		shell.Size = offset(size.X, size.Y)
		shell.Position = offset(pos.X, pos.Y)
		canvas.GroupTransparency = 0
		shell.BackgroundTransparency = 0
		shellStroke.Transparency = 0

		puck.Position = offset(target.X, target.Y)
		puckScale.Scale = 0.6
		puckLetter.TextTransparency = 1
		puck.Visible = true
		tw(puckScale, MOTION.knob, { Scale = 1 })
		tw(puckLetter, MOTION.enter, { TextTransparency = 0 })
		animatingMinimize = false
	end)
end

local function doRestore()
	if not minimized or animatingMinimize then return end
	animatingMinimize = true
	minimized = false

	local pPos = Vector2.new(puck.Position.X.Offset, puck.Position.Y.Offset)
	puckHome = pPos

	tw(puckLetter, MOTION.exit, { TextTransparency = 1 })
	tw(puckScale, TweenInfo.new(0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Scale = 0.6 })

	task.delay(0.13, function()
		puck.Visible = false
		puckScale.Scale = 1

		local vp = viewport()
		local size = shellRect.size
		local goal = fullscreen and shellRect.pos or Vector2.new(
			math.clamp(px(pPos.X + PUCK / 2 - size.X / 2), 8, math.max(8, px(vp.X - size.X - 8))),
			math.clamp(px(pPos.Y + PUCK / 2 - size.Y / 2), 8, math.max(8, px(vp.Y - size.Y - 8)))
		)

		shell.Size = offset(size.X * 0.94, size.Y * 0.94)

-- Created by and originally uploaded by https://robloxscripts.com/user/Fleece

		shell.Position = offset(goal.X + size.X * 0.03, goal.Y + size.Y * 0.03)
		shellCorner.CornerRadius = UDim.new(0, 14)
		canvas.GroupTransparency = 1
		shell.BackgroundTransparency = 1
		shellStroke.Transparency = 1
		shell.Visible = true

		tw(shell, TweenInfo.new(RES_MOVE, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Size = offset(size.X, size.Y),
			Position = offset(goal.X, goal.Y),
		})
		chromeFadeTo(0, 0.16)
		local done = shellFadeTo(0, RES_FADE)
		done.Completed:Connect(function() animatingMinimize = false end)
	end)
end

do
	local function clampPuck(x, y)
		local vp = viewport()
		return Vector2.new(
			math.clamp(px(x), 0, math.max(0, px(vp.X - PUCK))),
			math.clamp(px(y), 0, math.max(0, px(vp.Y - PUCK)))
		)
	end

	local moved, startPos, startMouse = false, nil, nil

	puck.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			moved = false
			startPos = Vector2.new(puck.Position.X.Offset, puck.Position.Y.Offset)
			startMouse = input.Position
			tw(puck, MOTION.micro, { BackgroundColor3 = T.hover })
			activeDrag = {
				move = function(p)
					local d = p - startMouse
					if math.abs(d.X) > 3 or math.abs(d.Y) > 3 then moved = true end
					local c = clampPuck(startPos.X + d.X, startPos.Y + d.Y)
					puck.Position = offset(c.X, c.Y)
				end,
				finish = function()
					tw(puck, MOTION.micro, { BackgroundColor3 = T.bg })
					local c = clampPuck(puck.Position.X.Offset, puck.Position.Y.Offset)
					puck.Position = offset(c.X, c.Y)
					puckHome = c
					if not moved then doRestore() end
				end,
			}
		end
	end)

	puck.MouseEnter:Connect(function()
		if activeDrag then return end
		tw(puckStroke, MOTION.hover, { Color = T.hoverLine })
		tw(puckScale, MOTION.hover, { Scale = 1.08 })
	end)
	puck.MouseLeave:Connect(function()
		if activeDrag then return end
		tw(puckStroke, MOTION.hover, { Color = T.shellLine })
		tw(puckScale, MOTION.hover, { Scale = 1 })
	end)

	rootMaid:give(Workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
		if not minimized then return end
		local c = clampPuck(puck.Position.X.Offset, puck.Position.Y.Offset)
		puck.Position = offset(c.X, c.Y)
		puckHome = c
	end))
end

local function refreshVisibility(animate)
	local q = searchBox.Text:lower():gsub("^%s+", ""):gsub("%s+$", "")
	searching = q ~= ""

	local shown, index = 0, 0
	for _, entry in ipairs(cards) do
		local visible
		if searching then
			visible = entry.q:find(q, 1, true) ~= nil
		else
			visible = entry.tab == currentTab
		end

		if entry.chip then entry.chip.Visible = searching end

		if visible then
			shown += 1
			if animate and entry.fade then

				setFade(entry.fade, 1)
				entry.frame.Visible = true
				tweenFade(entry.fade, 0, MOTION.card, index * 0.028)
				index += 1
			else
				if entry.fade then setFade(entry.fade, 0) end
				entry.frame.Visible = true
			end
		else
			entry.frame.Visible = false
		end
	end

	emptyState.Visible = shown == 0

	if animate then
		page.Position = offset(0, PAGE_Y + 10)
		tw(page, MOTION.card, { Position = offset(0, PAGE_Y) })
	end

	return shown
end

local function makeTabButton(name, idx)
	local b = new("TextButton", {
		BackgroundColor3 = T.optHover,
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, TAB_H),
		Text = "",

		Size = UDim2.fromScale(1, 1),
		Font = FONT_MED,
		Text = name,
		TextSize = 16,
		TextColor3 = T.muted,
		TextXAlignment = Enum.TextXAlignment.Center,
		TextYAlignment = Enum.TextYAlignment.Center,
		ZIndex = 13,
		Parent = b,
	})

	b.MouseEnter:Connect(function()
		if currentTab ~= name then
			tw(b, MOTION.hover, { BackgroundTransparency = 0.6 })
			tw(lbl, MOTION.hover, { TextColor3 = T.label })
		end
	end)
	b.MouseLeave:Connect(function()
		if currentTab ~= name then
			tw(b, MOTION.hover, { BackgroundTransparency = 1 })
			tw(lbl, MOTION.hover, { TextColor3 = T.muted })
		end
	end)
	b.Activated:Connect(function()
		if currentTab == name then return end
		if closeActivePopup then closeActivePopup() end
		currentTab = name
		if searchBox.Text ~= "" then searchBox.Text = "" end
		for tabName, entry in pairs(tabButtons) do
			local on = tabName == name
			tw(entry.btn, MOTION.hover, { BackgroundTransparency = 1 })
			tw(entry.lbl, MOTION.hover, { TextColor3 = on and T.white or T.muted })
		end
		movePill(idx)
		page.CanvasPosition = Vector2.new()
		refreshVisibility(true)
	end)

	tabButtons[name] = { btn = b, lbl = lbl, index = idx }

	task.defer(function()
		if not lbl.Parent then return end
		local slack = lbl.AbsoluteSize.X - lbl.TextBounds.X
		if slack >= 0 and slack % 2 == 1 then
			lbl.Size = UDim2.new(1, -1, 1, 0)
		end
	end)

	return b
end

	local function paintSelection()
		for _, row in ipairs(optionRows) do
			local selected = row.item.value == value
			row.selected = selected
			row.btn.BackgroundTransparency = selected and 0.55 or 1
			row.label.TextColor3 = row.item.locked and T.tag

	local function buildOptions()
		for _, row in ipairs(optionRows) do row.btn:Destroy() end

					Text = it.note,

					TextXAlignment = Enum.TextXAlignment.Right,
					TextTruncate = Enum.TextTruncate.AtEnd,
					ZIndex = 523,
					Parent = ob,
				})
			end

			local row = { btn = ob, label = label, item = it, selected = false }
			optionRows[i] = row

			ob.MouseEnter:Connect(function()
				tw(ob, MOTION.micro, { BackgroundTransparency = 0 })
			end)
			ob.MouseLeave:Connect(function()
				tw(ob, MOTION.micro, { BackgroundTransparency = row.selected and 0.55 or 1 })
			end)
			ob.Activated:Connect(function()
				if it.locked then
					notify(("%s is unavailable, %s."):format(it.text, it.note or "unsupported"), "bad")
					return
				end
				value = it.value
				lbl.Text = it.text
				paintSelection()
				if closeActivePopup then closeActivePopup() end
				if onChange then
					local ok, err = pcall(onChange, value)
					if not ok then notify("Error: " .. tostring(err), "bad", 6) end
				end
			end)
		end

		paintSelection()
		popupFade = buildFade(popup)
	end
	buildOptions()

	local open = false
	local wheelConn = nil

	local function close()
		if not open then return end
		open = false
		closeActivePopup = nil
		page.ScrollingEnabled = true
		if wheelConn then
			wheelConn:Disconnect()
			wheelConn = nil
		end

		tw(chev, MOTION.enter, { Rotation = 0 })
		tweenFade(popupFade, 1, MOTION.exit)
		local cur = popup.Position
		local out = tw(popup, MOTION.exit, {
			Position = offset(cur.X.Offset, cur.Y.Offset - 6),
		})
		out.Completed:Connect(function()
			if open then return end
			popup.Visible = false
			if not closeActivePopup then overlay.Visible = false end
		end)
	end

	local function show()
		if closeActivePopup then closeActivePopup() end
		open = true
		closeActivePopup = close

		local contentH = #items * (OPT_H + 2) + 8
		local height = math.min(contentH, MAX_POPUP_H)
		local width = btn.AbsoluteSize.X

		local ap = btn.AbsolutePosition - overlay.AbsolutePosition
		local overlayH = overlay.AbsoluteSize.Y

		local belowY = ap.Y + btn.AbsoluteSize.Y + 4
		local finalY, startY
		if belowY + height > overlayH - 10 then
			finalY = math.max(10, ap.Y - height - 4)
			startY = finalY + 10
		else
			finalY = belowY
			startY = finalY - 10
		end

		popup.Size = offset(width, height)
		popup.Position = offset(ap.X, startY)
		popup.Visible = true
		overlay.Visible = true
		scroller.CanvasPosition = Vector2.new()

		setFade(popupFade, 1)
		tweenFade(popupFade, 0, MOTION.enter)
		tw(chev, MOTION.enter, { Rotation = 180 })
		tw(popup, MOTION.enter, { Position = offset(ap.X, finalY) })

		page.ScrollingEnabled = false
		wheelConn = UserInputService.InputChanged:Connect(function(input)
			if input.UserInputType ~= Enum.UserInputType.MouseWheel then return end
			local m = UserInputService:GetMouseLocation()
			local p, s = popup.AbsolutePosition, popup.AbsoluteSize
			local inside = m.X >= p.X and m.X <= p.X + s.X and m.Y >= p.Y and m.Y <= p.Y + s.Y
			if inside then
				local maxScroll = math.max(0, scroller.AbsoluteCanvasSize.Y - scroller.AbsoluteSize.Y)
				scroller.CanvasPosition = Vector2.new(0, math.clamp(
					scroller.CanvasPosition.Y - input.Position.Z * 42, 0, maxScroll
				))
			else
				close()
			end
		end)
	end

	btn.Activated:Connect(function()
		if open then close() else show() end
	end)

	return {
		instance = btn,
		get = function() return value end,
		set = function(v, fire)
			value = v
			lbl.Text = textFor(v)
			paintSelection()
			if fire and onChange then pcall(onChange, v) end
		end,
		setItems = function(list, keepValue)
			items = list
			if not keepValue then
				local stillValid = false
				for _, it in ipairs(list) do
					if it.value == value then stillValid = true break end
				end
				if not stillValid and list[1] then
					value = list[1].value
					lbl.Text = list[1].text
					if onChange then pcall(onChange, value) end
				end
			end
			buildOptions()
		end,
		close = close,
	}
end

local function Toggle(parent, cfg)
	local state = cfg.value and true or false

	local btn = new("TextButton", {
		BackgroundColor3 = state and T.white or T.field,
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 0, 0.5, 0),
		Size = offset(44, 24),
		Text = "",
		AutoButtonColor = false,
		BorderSizePixel = 0,
		ZIndex = cfg.zindex or 14,
		Parent = parent,
	})
	corner(btn, 12)
	local s = stroke(btn, state and T.white or T.fieldLine)

	local knob = new("Frame", {
		BackgroundColor3 = state and T.black or T.muted,
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, state and 23 or 3, 0.5, 0),
		Size = offset(18, 18),
		BorderSizePixel = 0,
		ZIndex = (cfg.zindex or 14) + 1,
		Parent = btn,
	})
	corner(knob, 9)

	local api = {}

	local function render(v, animate)
		state = v and true or false
		local info = animate == false and MOTION.micro or MOTION.knob
		tw(btn, MOTION.hover, { BackgroundColor3 = state and T.white or T.field })
		tw(s, MOTION.hover, { Color = state and T.white or T.fieldLine })
		tw(knob, info, {
			Position = UDim2.new(0, state and 23 or 3, 0.5, 0),
			BackgroundColor3 = state and T.black or T.muted,
		})
	end

	function api.get() return state end
	function api.set(v) render(v) end

	btn.MouseButton1Down:Connect(function()
		tw(knob, MOTION.micro, { Size = offset(20, 18) })
	end)

	btn.Activated:Connect(function()
		tw(knob, MOTION.knob, { Size = offset(18, 18) })

		local want = not state
		if not cfg.onChange then
			render(want)
			return
		end
		local ok, result = pcall(cfg.onChange, want)
		if not ok then
			notify("Error: " .. tostring(result), "bad", 6)
			render(state)
			return
		end
		if result == false then
			render(state)
		else
			render(want)
		end
	end)

	api.instance = btn
	return api
end

local function Slider(parent, cfg)
	local min, max = cfg.min or 0, cfg.max or 100
	local step = cfg.step or 1
	local value = math.clamp(cfg.value or min, min, max)

	local holder = new("Frame", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 0, 0.5, 0),
		Size = UDim2.new(1, 0, 0, 24),
		ZIndex = cfg.zindex or 14,
		Parent = parent,
	})

	local valueLabel = new("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, 0, 0, 0),
		Size = UDim2.new(0, 56, 1, 0),
		Font = FONT_MONO,
		Text = tostring(value),
		TextSize = 13,
		TextColor3 = T.muted,
		TextXAlignment = Enum.TextXAlignment.Right,
		ZIndex = (cfg.zindex or 14) + 1,
		Parent = holder,
	})

	local track = new("Frame", {
		BackgroundColor3 = T.track,
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 0, 0.5, 0),
		Size = UDim2.new(1, -66, 0, 6),
		BorderSizePixel = 0,
		ZIndex = (cfg.zindex or 14) + 1,
		Parent = holder,
	})
	corner(track, 3)

	local hit = new("TextButton", {
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.new(1, 14, 0, 26),
		ZIndex = (cfg.zindex or 14) + 4,
		Parent = track,
	})

	local fill = new("Frame", {
		BackgroundColor3 = T.white,
		Size = UDim2.fromScale((value - min) / (max - min), 1),
		BorderSizePixel = 0,
		ZIndex = (cfg.zindex or 14) + 2,
		Parent = track,
	})
	corner(fill, 3)

	local knob = new("Frame", {
		BackgroundColor3 = T.white,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new((value - min) / (max - min), 0, 0.5, 0),
		Size = offset(14, 14),
		BorderSizePixel = 0,
		ZIndex = (cfg.zindex or 14) + 3,
		Parent = track,
	})
	corner(knob, 7)

	local function fmt(v)
		if step < 1 then return string.format("%.2f", v) end
		return tostring(math.floor(v + 0.5))
	end

	local dragging = false

	local function apply(v, fire, animate)
		if type(v) ~= "number" or v ~= v then return end
		v = math.clamp(v, min, max)
		v = math.floor(v / step + 0.5) * step
		value = math.clamp(v, min, max)
		local alpha = (value - min) / (max - min)

		local width = track.AbsoluteSize.X
		local filled = px(alpha * width)
		local goalFill = width > 0 and UDim2.fromOffset(filled, 6) or UDim2.fromScale(alpha, 1)
		local goalKnob = width > 0 and UDim2.new(0, filled, 0.5, 0) or UDim2.new(alpha, 0, 0.5, 0)

		if animate then
			tw(fill, MOTION.enter, { Size = goalFill })
			tw(knob, MOTION.enter, { Position = goalKnob })
		else
			fill.Size = goalFill
			knob.Position = goalKnob
		end

		valueLabel.Text = fmt(value)
		if fire and cfg.onChange then
			local ok, err = pcall(cfg.onChange, value)
			if not ok then local _ = err end
		end
	end
	apply(value, false, false)

	rootMaid:give(track:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
		apply(value, false, false)
	end))

	local function fromX(x)
		local w = track.AbsoluteSize.X
		if w <= 0 then return end
		local alpha = math.clamp((x - track.AbsolutePosition.X) / w, 0, 1)
		apply(min + alpha * (max - min), true, false)
	end

	hit.MouseEnter:Connect(function()
		if not dragging then tw(knob, MOTION.hover, { Size = offset(16, 16) }) end
	end)
	hit.MouseLeave:Connect(function()
		if not dragging then tw(knob, MOTION.hover, { Size = offset(14, 14) }) end
	end)

	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			if closeActivePopup then closeActivePopup() end
			dragging = true
			tw(knob, MOTION.knob, { Size = offset(18, 18) })
			fromX(input.Position.X)
			activeDrag = {
				move = function(p) fromX(p.X) end,
				finish = function()
					dragging = false
					tw(knob, MOTION.enter, { Size = offset(14, 14) })
				end,
			}
		end
	end)

	return {
		instance = holder,
		get = function() return value end,
		set = function(v, fire) apply(v, fire, true) end,
	}
end

		ZIndex = cfg.zindex or 14,
		Parent = parent,
	})

	local s = stroke(b, T.buttonLine)
	hoverable(b, T.button, T.hover, s, T.buttonLine, T.hoverLine)

		b.TextColor3 = key and T.title or T.tag
		tw(s, MOTION.hover, { Color = T.buttonLine })
	end

	b.Activated:Connect(function()
		if listening then listening.cancel() end
		b.Text = "press a key"
		b.TextColor3 = T.yellow
		tw(s, MOTION.hover, { Color = T.yellow })
		listening = {
			apply = function(code)
				listening = nil
				if code == Enum.KeyCode.Escape or code == Enum.KeyCode.Backspace then
					key = nil
				else
					key = code
				end
				render()
				if cfg.onChange then cfg.onChange(key) end
			end,
			cancel = function()
				listening = nil
				render()
			end,
		}
	end)

	api.instance = b
	api.get = function() return key end
	api.set = function(k) key = k render() end
	return api
end

local Runtime = { active = {}, state = {}, onChanged = nil }
local Features = {}
local featureOrder = {}

local Mobile

local everySeq = 0

local function ctxFor(id, methodDef)
	local maid = Maid.new()
	local restorer = Restorer()
	maid:give(function() restorer.restoreAll() end)

	local ctx = {
		maid = maid,
		s = Runtime.state[id].settings,
		method = methodDef,
		notify = notify,
		restorer = restorer,
	}

	function ctx:every(bucket, fn)
		everySeq += 1
		local key = ("%s:%s:%d"):format(id, bucket, everySeq)
		Scheduler.add(bucket, key, fn, function(err)
			notify(("%s stopped, %s"):format(Features[id].name, tostring(err)), "bad", 7)
			task.spawn(function()
				Runtime.setEnabled(id, false)
				if Runtime.onChanged then Runtime.onChanged(id, false) end
			end)
		end)
		maid:give(function() Scheduler.remove(bucket, key) end)
	end

	function ctx:throttled(bucket, interval, fn)
		local acc = interval
		self:every(bucket, function(dt)
			acc += dt
			if acc >= interval then
				local used = acc
				acc = 0
				fn(used)
			end
		end)
	end

	function ctx:restore(obj, prop)
		restorer.set(obj, prop, obj[prop])
	end

	function ctx:set(obj, prop, value)
		restorer.set(obj, prop, value)
	end

	return ctx
end

function Runtime.register(def)
	Features[def.id] = def
	table.insert(featureOrder, def.id)
	Runtime.state[def.id] = {
		enabled = false,
		method = def.methods[1].id,
		settings = {},
		keybind = nil,
	}
	for key, meta in pairs(def.settings or {}) do
		Runtime.state[def.id].settings[key] = meta.default
	end
end

local function methodById(def, mid)
	for _, m in ipairs(def.methods) do
		if m.id == mid then return m end
	end
	return def.methods[1]
end

function Runtime.stop(id)
	local live = Runtime.active[id]
	if not live then return end
	Runtime.active[id] = nil
	if live.method.stop then
		local ok, err = pcall(live.method.stop, live.ctx)
		if not ok then local _ = err end
	end
	live.ctx.maid:clean()
end

function Runtime.start(id)
	local def = Features[id]
	if not def then return false end
	local st = Runtime.state[id]
	local m = methodById(def, st.method)

	local ok, missing = capable(m.needs)
	if not ok then
		notify(("%s > %s needs %s(), which this executor doesn't have."):format(def.name, m.name, missing), "bad", 6)
		return false
	end

	if m.requiresChar ~= false and not alive() then
		notify(("%s needs a living character."):format(def.name), "warn")
		return false
	end

	for _, otherId in ipairs(def.conflicts or {}) do
		if Runtime.active[otherId] then
			notifyThrottled("conflict:" .. id, 8,
				("%s and %s both control the same properties, expect odd behaviour.")
					:format(def.name, Features[otherId] and Features[otherId].name or otherId), "warn")
		end
	end

	Runtime.stop(id)

	local ctx = ctxFor(id, m)
	local started, err = pcall(m.start, ctx)
	if not started then
		ctx.maid:clean()
		notify(("%s > %s failed: %s"):format(def.name, m.name, tostring(err)), "bad", 7)
		return false
	end

	if def.mobileVertical and Mobile then
		Mobile.request(true)
		ctx.maid:give(function() Mobile.request(false) end)
	end

	Runtime.active[id] = { ctx = ctx, method = m }
	return true
end

function Runtime.setEnabled(id, on)
	local st = Runtime.state[id]
	if not st then return false end
	if on then
		if Runtime.start(id) then
			st.enabled = true
			return true
		end
		st.enabled = false
		return false
	end
	Runtime.stop(id)
	st.enabled = false
	return true
end

function Runtime.setMethod(id, mid)
	local st = Runtime.state[id]
	if not st then return false end
	local wasEnabled = st.enabled
	st.method = mid
	if wasEnabled then
		Runtime.stop(id)
		if not Runtime.start(id) then
			st.enabled = false
			return false
		end

-- Created by and originally uploaded by https://robloxscripts.com/user/Fleece

	end
	return true
end

table.insert(Char.onRespawn, function()
	local wanted = {}
	for id, st in pairs(Runtime.state) do
		if st.enabled then table.insert(wanted, id) end
	end
	if #wanted == 0 then return end

	if not awaitAlive(8) then
		for _, id in ipairs(wanted) do
			Runtime.setEnabled(id, false)
			if Runtime.onChanged then Runtime.onChanged(id, false) end
		end
		notify("Character never became ready, features were turned off.", "warn", 6)
		return
	end

	task.wait(0.25)
	for _, id in ipairs(wanted) do
		Runtime.stop(id)
		if not Runtime.start(id) then
			Runtime.state[id].enabled = false
			if Runtime.onChanged then Runtime.onChanged(id, false) end
		end
	end
end)

local function moveDir(includeVertical)
	local cam = camera()
	if not cam then return Vector3.zero end
	local cf = cam.CFrame
	local d = Vector3.zero

	if not typing() then
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then d += cf.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then d -= cf.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then d -= cf.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then d += cf.RightVector end
		if includeVertical then
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then d += Vector3.yAxis end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then d -= Vector3.yAxis end
		end
	end

	if d.Magnitude < 1e-4 and Char.hum then
		local md = Char.hum.MoveDirection
		if md.Magnitude > 0 then

			local flat = flatLook(cf.LookVector)
			local right = Vector3.new(flat.Z, 0, -flat.X)
			local fwd = md:Dot(flat)
			local side = md:Dot(right)
			d = cf.LookVector * fwd + cf.RightVector * side
		end
	end

	if includeVertical and Mobile.vertical ~= 0 then
		d += Vector3.yAxis * Mobile.vertical
	end

	if d.Magnitude < 1e-4 then return Vector3.zero end
	return safeVector(d.Unit)
end

local function nearestPlayer(maxDist)
	local best, bestDist = nil, maxDist or math.huge
	if not Char.root then return nil end
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				local d = (r.Position - Char.root.Position).Magnitude
				if d < bestDist then best, bestDist = p, d end
			end
		end
	end
	return best
end

local function playerItems(includeNearest)
	local list = {}
	if includeNearest then
		table.insert(list, { value = "@nearest", text = "Nearest player" })
	end
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP then
			local label = p.DisplayName ~= p.Name and (p.DisplayName .. " (" .. p.Name .. ")") or p.Name
			table.insert(list, { value = p.Name, text = label })
		end
	end
	if #list == 0 then
		table.insert(list, { value = "@none", text = "No other players" })
	end
	return list
end

local function resolveTarget(value)
	if value == "@nearest" then return nearestPlayer() end
	if value == "@none" or not value then return nil end
	return Players:FindFirstChild(value)
end

end

local AIM_PART_SETS = {
	head  = { "Head" },
	torso = { "UpperTorso", "Torso" },
	root  = { "HumanoidRootPart" },
}

local function aimPartOf(char, key)
	if key == "closest" then
		local best, bestD
		local cursor = UserInputService:GetMouseLocation()
		local cam = camera()
		for _, list in pairs(AIM_PART_SETS) do
			for _, n in ipairs(list) do
				local part = char:FindFirstChild(n)
				if part and part:IsA("BasePart") then
					local sp = cam:WorldToViewportPoint(part.Position)
					if sp.Z > 0 then
						local d = (Vector2.new(sp.X, sp.Y) - cursor).Magnitude
						if not bestD or d < bestD then best, bestD = part, d end
					end
				end
			end
		end
		return best or char:FindFirstChild("HumanoidRootPart")
	end
	for _, n in ipairs(AIM_PART_SETS[key] or AIM_PART_SETS.head) do
		local part = char:FindFirstChild(n)
		if part and part:IsA("BasePart") then return part end
	end
	return char:FindFirstChild("HumanoidRootPart")
end

local function hasLineOfSight(part)
	local cam = camera()
	if not cam then return false end
	local origin = cam.CFrame.Position
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { Char.model, part.Parent }
	params.IgnoreWater = true
	local hit = Workspace:Raycast(origin, part.Position - origin, params)
	return hit == nil
end

local function aimHoldActive(ctx)
	if not ctx.s.holdaim then return true end
	if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return true end
	return UserInputService.TouchEnabled and Mobile and Mobile.vertical ~= 0
end

local function findAimTarget(ctx)
	local cam = camera()
	if not cam or not alive() then return nil end
	local cursor = UserInputService:GetMouseLocation()
	local best, bestD

	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and p.Character then
			local skip = ctx.s.teamcheck and p.Team == LP.Team
			local hum = p.Character:FindFirstChildOfClass("Humanoid")
			if not skip and hum and hum.Health > 0 then
				local part = aimPartOf(p.Character, ctx.s.part)
				if part then
					local sp = cam:WorldToViewportPoint(part.Position)
					if sp.Z > 0 then
						local d = (Vector2.new(sp.X, sp.Y) - cursor).Magnitude
						if d <= ctx.s.fov and (not bestD or d < bestD) then
							if not ctx.s.wallcheck or hasLineOfSight(part) then
								best, bestD = part, d
							end
						end
					end
				end
			end
		end
	end
	return best
end

local function attachFovCircle(ctx)
	local ring = ctx.maid:give(new("Frame", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = offset(ctx.s.fov * 2, ctx.s.fov * 2),
		Visible = false,
		ZIndex = 450,
		Parent = screen,
	}))
	new("UICorner", { CornerRadius = UDim.new(1, 0), Parent = ring })
	local ringStroke = new("UIStroke", {
		Color = T.white,
		Thickness = 1,
		Transparency = 0.45,
		Parent = ring,
	})

	ctx:every("render", function()
		if not ctx.s.showfov then
			ring.Visible = false
			return
		end
		local m = UserInputService:GetMouseLocation()
		ring.Visible = true
		ring.Size = offset(ctx.s.fov * 2, ctx.s.fov * 2)
		ring.Position = offset(m.X, m.Y)
		ringStroke.Color = ctx.locked and T.red or T.white
	end)
end

local tpMethod = "direct"
local waypoints = {}

local function doTeleport(target)
	if not alive() then
		notify("No character to teleport.", "bad")
		return false
	end
	local goal = typeof(target) == "CFrame" and target or CFrame.new(target)
	if not isFinite(goal.Position.X) or not isFinite(goal.Position.Y) or not isFinite(goal.Position.Z) then
		notify("That destination is not a valid position.", "bad")
		return false
	end

	if tpMethod == "direct" then
		Char.root.CFrame = goal

	elseif tpMethod == "chunk" then
		task.spawn(function()
			local step, guard = 60, 0
			while alive() and guard < 500 do
				local root = Char.root
				local delta = goal.Position - root.Position
				if delta.Magnitude <= step then break end
				guard += 1
				root.CFrame = root.CFrame + delta.Unit * step
				root.AssemblyLinearVelocity = Vector3.zero
				task.wait()
			end
			if alive() then Char.root.CFrame = goal end
		end)

	elseif tpMethod == "tween" then
		local dist = (Char.root.Position - goal.Position).Magnitude
		local duration = math.clamp(dist / 120, 0.15, 12)
		TweenService:Create(Char.root, TweenInfo.new(duration, Enum.EasingStyle.Linear),
			{ CFrame = goal }):Play()

	elseif tpMethod == "seat" then
		local seat = Char.hum and Char.hum.SeatPart
		if seat then
			seat.CFrame = goal * CFrame.new(0, 2, 0)
		else
			Char.root.CFrame = goal
			notify("Not seated, used a direct write instead.", "warn")
		end
	end
	return true
end

do
	local card = makeCard("Server", "Anti-Kick / Anti-Teleport", "Blocks the common eject paths.")

	local blockKick, blockTp = false, false
	local hookMaid = Maid.new()
	rootMaid:give(hookMaid)
	local hookInstalled = false

	local function syncHook()
		local wanted = blockKick or blockTp
		if wanted == hookInstalled then return end
		if wanted then
			local original
			original = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
				local method = getnamecallmethod()
				if not checkcaller() then
					if blockKick and method == "Kick" and self == LP then return nil end
					if blockTp and self == TeleportService
						and (method == "Teleport" or method == "TeleportToPlaceInstance"
							or method == "TeleportToPrivateServer") then
						return nil
					end
				end
				return original(self, ...)
			end))
			hookMaid:give(function()
				pcall(function() hookmetamethod(game, "__namecall", original) end)
			end)
			hookInstalled = true
		else
			hookMaid:clean()
			hookInstalled = false
		end
	end

	local hookNeeds = { "hookmetamethod", "newcclosure", "checkcaller", "getnamecallmethod" }
	local canHook, missingHook = capable(hookNeeds)

	local function makeBlockToggle(label, orderIdx, setter)
		local slot = labelRow(card, label, orderIdx, 28)
		Toggle(slot, {
			zindex = 15,
			onChange = function(v)
				if v and not canHook then
					notify(("Needs %s(), which this executor doesn't have."):format(missingHook), "bad")
					return false
				end
				setter(v)
				syncHook()
				notify(("%s %s."):format(label, v and "enabled" or "disabled"))
				return true
			end,
		})
	end

		end,
	})

	Button(row, {
		text = "Load", order = 2, size = UDim2.new(0.33, -6, 0, 34),
		onClick = function()
			local data, err = Config.load()
			if not data then
				return notify(err and ("Load failed: " .. err) or "No saved config for this place.", "warn", 5)
			end

			local applied, failed = 0, 0
			for id, saved in pairs(data.features or {}) do
				local st, h = Runtime.state[id], uiHandles[id]
				if st and h then
					for k, v in pairs(saved.settings or {}) do
						if st.settings[k] ~= nil then
							st.settings[k] = v
							local widget = h.settings[k]
							if widget and widget.set then pcall(widget.set, v, false) end
						end
					end

					if saved.method then

						local valid = false
						for _, m in ipairs(Features[id].methods) do
							if m.id == saved.method and capable(m.needs) then valid = true break end
						end
						if valid then
							st.method = saved.method
							h.method.set(saved.method, false)
						end
					end

					if saved.keybind then

						local okKey, code = pcall(function() return Enum.KeyCode[saved.keybind] end)
						if okKey and code then
							st.keybind = saved.keybind
							featureBinds[code] = id
							h.keybind.set(code)
						end
					end

					if saved.enabled and not st.enabled then
						if Runtime.setEnabled(id, true) then
							h.toggle.set(true)
							applied += 1
						else
							h.toggle.set(false)
							failed += 1
						end
					elseif not saved.enabled and st.enabled then
						Runtime.setEnabled(id, false)
						h.toggle.set(false)
					end
				end
			end

			if type(data.waypoints) == "table" then
				waypoints = data.waypoints
				redrawWaypoints()
			end
			if data.tpMethod then tpMethod = data.tpMethod end
			if data.uiToggleKey then
				local okKey, code = pcall(function() return Enum.KeyCode[data.uiToggleKey] end)
				if okKey and code then uiToggleKey = code end
			end

			notify(failed > 0
				and ("Config loaded, %d on, %d could not start."):format(applied, failed)
				or "Config loaded.", failed > 0 and "warn" or "good", 5)
		end,
	})

	Button(row, {
		text = "Turn all off", order = 3, size = UDim2.new(0.33, -6, 0, 34),
		onClick = function()
			local n = 0
			for id, st in pairs(Runtime.state) do
				if st.enabled then
					Runtime.setEnabled(id, false)
					local h = uiHandles[id]
					if h then h.toggle.set(false) end
					n += 1
				end
			end
			notify(n > 0 and ("Turned off %d feature%s."):format(n, n == 1 and "" or "s")
				or "Nothing was on.")
		end,
	})
end

do
	local function snapRow(row, layout)
		local kids = {}
		for _, c in ipairs(row:GetChildren()) do
			if c:IsA("GuiObject") then table.insert(kids, c) end
		end
		if #kids == 0 then return end
		table.sort(kids, function(a, b) return a.LayoutOrder < b.LayoutOrder end)

		local gap = layout.Padding.Offset
		local total = row.AbsoluteSize.X - gap * (#kids - 1)
		if total <= 0 then return end
		local base = math.floor(total / #kids)
		local extra = total - base * #kids
		for i, c in ipairs(kids) do
			c.Size = UDim2.fromOffset(base + (i <= extra and 1 or 0), c.Size.Y.Offset)
		end
	end

	for _, entry in ipairs(cards) do
		for _, d in ipairs(entry.frame:GetDescendants()) do
			if d:IsA("UIListLayout") and d.FillDirection == Enum.FillDirection.Horizontal then
				local row = d.Parent
				local allScaled = true
				local any = false
				for _, c in ipairs(row:GetChildren()) do
					if c:IsA("GuiObject") then
						any = true
						if c.Size.X.Scale == 0 then allScaled = false end
					end
				end
				if any and allScaled then
					local function relayout() snapRow(row, d) end
					rootMaid:give(row:GetPropertyChangedSignal("AbsoluteSize"):Connect(relayout))
					task.defer(relayout)
				end
			end
		end
	end
end

local unloaded = false

local function unload()
	if unloaded then return end
	unloaded = true

	for id in pairs(Runtime.state) do pcall(Runtime.stop, id) end
	Scheduler.stopAll()

	for id, st in pairs(Runtime.state) do
		local def = Features[id]
		st.enabled = false
		st.keybind = nil
		st.method = def and def.methods[1].id or st.method
		for key, meta in pairs((def and def.settings) or {}) do
			st.settings[key] = meta.default
		end
	end

	pcall(function() charMaid:destroy() end)
	pcall(function() rootMaid:destroy() end)
	pcall(function() screen:Destroy() end)

	for inst in pairs(spawned) do
		pcall(function()
			if inst.Parent ~= nil then inst:Destroy() end
		end)
	end
	table.clear(spawned)

	table.clear(featureBinds)
	table.clear(uiHandles)
	table.clear(Runtime.active)
	table.clear(playerDropdowns)
	table.clear(logLines)
	table.clear(lastNotifyAt)
	table.clear(Char.onRespawn)
	Char.model, Char.hum, Char.root = nil, nil, nil

	local registry = getgenv().__fleeceRegistry
	if type(registry) == "table" then
		for i = #registry, 1, -1 do
			if registry[i] == unload then table.remove(registry, i) end
		end
		if #registry == 0 then getgenv().__fleeceRegistry = nil end
	end
	if getgenv()[UNLOAD_KEY] == unload then getgenv()[UNLOAD_KEY] = nil end
	getgenv().VertexUnload = nil
end

getgenv()[UNLOAD_KEY] = unload
getgenv().__fleeceRegistry = getgenv().__fleeceRegistry or {}
table.insert(getgenv().__fleeceRegistry, unload)
