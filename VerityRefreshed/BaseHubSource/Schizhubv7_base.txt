    for _, o in ipairs(ColorPalette) do

-- THEME APPLY + RECOLOR SWEEP
-- ============================================================

-- Background/text compatibility sweep: any element still painted with the
-- previous gray/text tones gets moved to the new ones, so a custom background
-- or text color recolors the whole already-built window in one pass. The
-- previous tone set is parked on COLOR.__prev (a field, not a new local).
-- V6: the sweep also applies the Dark Glass translucency (BUILD_VARIANT ==
-- "glass") to every role-colored surface, so panels built at any time --
-- including floating panels opened later -- keep the glass look on theme
-- switches. recolorTree() is exposed at chunk scope so late-built panels
-- can be re-tinted on open.
local function recolorTree()
    local stack = {HUB.frame}
    while #stack > 0 do
        local obj = table.remove(stack)
        -- Guarded reads: not every instance type exposes BackgroundColor3
        -- or TextColor3 (UIStroke, UIListLayout, UIScale, ...), and Roblox
        -- throws on invalid member access, which would kill the whole
        -- recolor pass. Only match objects that actually have the member.
        local okB, bc = pcall(function() return obj.BackgroundColor3 end)
        if okB and bc then
            local prev = COLOR.__prev
            local finalBc = bc
            if prev then
                if bc == prev.panel then finalBc = COLOR.panel
                elseif bc == prev.playerBtn then finalBc = COLOR.playerBtn
                elseif bc == prev.switchOff then finalBc = COLOR.switchOff
                elseif bc == prev.bg then finalBc = COLOR.bg
                elseif bc == prev.titleBar then finalBc = COLOR.titleBar
                end
                if finalBc ~= bc then obj.BackgroundColor3 = finalBc end
            end
            if BUILD_VARIANT == "glass" then
                local tint = (finalBc == COLOR.bg and 0.30)
                    or (finalBc == COLOR.panel and 0.32)
                    or (finalBc == COLOR.playerBtn and 0.42)
                    or (finalBc == COLOR.switchOff and 0.55)
                    or (finalBc == COLOR.titleBar and 0.26)
                    or nil
                if tint then obj.BackgroundTransparency = tint end
            end
        end
        local okT, tc = pcall(function() return obj.TextColor3 end)
        if okT and tc and prev then
            if tc == prev.text then obj.TextColor3 = COLOR.text
            elseif tc == prev.textDim then obj.TextColor3 = COLOR.textDim
            end
        end
        for _, child in ipairs(obj:GetChildren()) do table.insert(stack, child) end
    end
end
local function refreshTheme()
    if HUB.frame then recolorTree() end
    for _, e in ipairs(themedElements) do
        if e.instance and e.instance.Parent then
            e.instance[e.property] = type(e.key) == "function" and e.key() or COLOR[e.key]
        end
    end
    for _, t in ipairs(switchTracks) do
        if t.track.Parent then
            t.track.BackgroundColor3 = t.get() and COLOR.accent or COLOR.switchOff
            if t.mark then t.mark.TextColor3 = Color3.fromRGB(235, 235, 235) end
        end
    end
    COLOR.__prev = {
        bg = COLOR.bg, panel = COLOR.panel, playerBtn = COLOR.playerBtn,
        switchOff = COLOR.switchOff, titleBar = COLOR.titleBar,
        text = COLOR.text, textDim = COLOR.textDim,
    }
end

-- Lighten (positive factor) or darken (negative factor) a color toward the
-- theme's light/dark variants.
local function shadeColor(col, factor)
    local r, g, b = col.R * 255, col.G * 255, col.B * 255
    if factor >= 0 then
        return Color3.fromRGB(r + (255 - r) * factor, g + (255 - g) * factor, b + (255 - b) * factor)
    end
    return Color3.fromRGB(r * (1 + factor), g * (1 + factor), b * (1 + factor))
end
-- Background compatibility: panels, player buttons, switch tracks and the
-- header bar all derive from the background (a fixed per-channel shift toward
-- light or dark), so a custom background never leaves the buttons looking odd.
-- The sweep that repaints already-built elements lives in refreshTheme().
local function deriveBackground(bg)
    COLOR.bg = bg
    COLOR.panel = Color3.fromRGB(math.clamp(bg.R * 255 + 6, 0, 255), math.clamp(bg.G * 255 + 6, 0, 255), math.clamp(bg.B * 255 + 7, 0, 255))
    COLOR.playerBtn = Color3.fromRGB(math.clamp(bg.R * 255 + 12, 0, 255), math.clamp(bg.G * 255 + 12, 0, 255), math.clamp(bg.B * 255 + 13, 0, 255))
    COLOR.switchOff = Color3.fromRGB(math.clamp(bg.R * 255 + 26, 0, 255), math.clamp(bg.G * 255 + 26, 0, 255), math.clamp(bg.B * 255 + 28, 0, 255))
    COLOR.titleBar = Color3.fromRGB(math.clamp(bg.R * 255 - 4, 0, 255), math.clamp(bg.G * 255 - 4, 0, 255), math.clamp(bg.B * 255 - 4, 0, 255))
end
local function deriveText(text)
    COLOR.text = text
    COLOR.textDim = Color3.fromRGB(math.clamp(text.R * 255 - 85, 0, 255), math.clamp(text.G * 255 - 85, 0, 255), math.clamp(text.B * 255 - 85, 0, 255))
end
-- All display theming is preset-based (no custom color slot): applying a theme
-- just swaps the accent family and derives the background/text compatibility
-- tones, then repaints everything already built.
local function applyTheme(preset)
    currentPreset = preset
    COLOR.accent, COLOR.accent2, COLOR.accent3 = preset.accent, preset.accent2, preset.accent3
    COLOR.accentLight, COLOR.accentDark = shadeColor(preset.accent, 0.4), shadeColor(preset.accent, -0.4)
    COLOR.accent2Light, COLOR.accent2Dark = shadeColor(preset.accent2, 0.4), shadeColor(preset.accent2, -0.4)
    COLOR.accent3Light, COLOR.accent3Dark = shadeColor(preset.accent3, 0.4), shadeColor(preset.accent3, -0.4)
    if preset.bg then deriveBackground(preset.bg) end
    if preset.text then deriveText(preset.text) end
    refreshTheme()
end

local themedFonts = {}
local currentFontPreset = FONT_PRESETS[1]
-- [FIX 2] Normalize the recorded base at bind time so objects bound before
-- their TextSize is meaningful can't poison applyTextScale.
-- [FIX 11] Guard against non-text instances: reading TextSize on a Frame
-- throws in real Roblox, so skip anything that is not a text object.
local function bindFont(obj, role)
    if not (obj:IsA("TextButton") or obj:IsA("TextLabel") or obj:IsA("TextBox")) then
        return
    end
    table.insert(themedFonts, {instance = obj, role = role, base = math.max(8, obj.TextSize or 12)})
end
local function applyFont(preset)
    currentFontPreset = preset
    for _, e in ipairs(themedFonts) do
        if e.instance and e.instance.Parent then
            e.instance.Font = (e.role == "bold") and preset.bold or preset.regular
        end
    end
end

-- size is rescaled by the preset's text factor, so bigger hubs read naturally.
local function applyTextScale(factor)
    for _, e in ipairs(themedFonts) do
        if e.instance and e.instance.Parent then
            e.instance.TextSize = math.max(8, math.floor((e.base or 12) * factor))
        end
    end
end
local function applySize(preset)
    currentSizePreset = preset
    TweenService:Create(hubUIScale, TweenInfo.new(0.15), {Scale = preset.scale}):Play()
    applyTextScale(preset.text or 1)
end

local function newInstance(className, props)
    local obj = Instance.new(className)
    for k, v in pairs(props) do obj[k] = v end
    return obj
end

-- Both helpers now return the created instance so the window shell can keep
-- handles on them for live border/corner restyling.
local function round(obj, radius)
    local corner = newInstance("UICorner", {CornerRadius = UDim.new(0, radius or 8), Parent = obj})
    return corner
end

local function stroke(obj, color, thickness)
    local edge = newInstance("UIStroke", {Color = color or COLOR.border, Thickness = thickness or 1, Parent = obj})
    return edge
end

        btn.BackgroundColor3 = shadeColor(restColor(), -0.16)
    end)
    btn.MouseLeave:Connect(function()
        btn.BackgroundColor3 = restColor()
    end)
end

-- ============================================================
-- TOAST NOTIFICATIONS (defined here because createColorPicker and
-- the command engine both use showToast; the label is built lazily
-- on first use so Part C does not depend on HUB.frame (Part D)).
-- ============================================================
local toastToken = 0
local function showToast(text, colorKey)
    if not HUB.toast or not HUB.toast.Parent then
        -- V6 design: rounded pill with a border instead of bare text.
        HUB.toast = newInstance("TextLabel", {
            Size = UDim2.new(1, -16, 0, 18),
            Position = UDim2.new(0, 8, 1, -22),
            BackgroundColor3 = COLOR.titleBar,
            BackgroundTransparency = 0,
            Text = "",
            TextTransparency = 1,
            TextColor3 = COLOR.warn,
            Font = currentFontPreset.bold,
            TextSize = 11,
            TextXAlignment = Enum.TextXAlignment.Center,
            Parent = (HUB.frame or HUB.screenGui),
            ZIndex = 40,
        })
        round(HUB.toast, 9)
        HUB.toastEdge = stroke(HUB.toast, COLOR.border, 1)
        bindThemed(HUB.toast, "BackgroundColor3", "titleBar")
        bindThemed(HUB.toastEdge, "Color", "border")
        bindFont(HUB.toast, "bold")
    end
    toastToken += 1
    local myToken = toastToken
    HUB.toast.Text = text
    HUB.toast.TextColor3 = (colorKey and COLOR[colorKey]) or COLOR.warn
    HUB.toast.TextTransparency = 0
    HUB.toast.BackgroundTransparency = 0
    if HUB.toastEdge then HUB.toastEdge.Transparency = 0 end
    HUB.toast.Visible = true
    task.delay(1.6, function()
        if myToken == toastToken then
            -- V7: fade the WHOLE pill (fill + text + border), then hide,
            -- so no ghost background lingers under the hub.
            TweenService:Create(HUB.toast, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
            TweenService:Create(HUB.toast, TweenInfo.new(0.4), {BackgroundTransparency = 1}):Play()
            if HUB.toastEdge then TweenService:Create(HUB.toastEdge, TweenInfo.new(0.4), {Transparency = 1}):Play() end
            task.delay(0.4, function()
                if myToken == toastToken then HUB.toast.Visible = false end
            end)
        end
    end)
end

local function createDropdown(toggleButton, labelPrefix, options, onSelect)
    toggleButton.MouseButton1Click:Connect(function()
        if activeOverlay then closeActiveOverlay() return end

        local pos, size = toggleButton.AbsolutePosition, toggleButton.AbsoluteSize
        -- V7: option rows + text grow with the hub size preset.
        local textScale = (currentSizePreset and currentSizePreset.text) or 1
        local optH = math.floor(24 * textScale)
        local listH = optH * #options
        local screenH = workspace.CurrentCamera.ViewportSize.Y
        local openUp = (pos.Y + size.Y + listH) > screenH
        local y = openUp and (pos.Y - listH) or (pos.Y + size.Y)

        local panel = newInstance("Frame", {
            Size = UDim2.new(0, size.X, 0, listH),
            Position = UDim2.new(0, pos.X, 0, y),
            BackgroundColor3 = COLOR.panel, ZIndex = 51, Parent = HUB.overlayLayer,
        })
        round(panel, 6)
        stroke(panel, COLOR.border, 1)
        newInstance("UIListLayout", {Parent = panel, SortOrder = Enum.SortOrder.LayoutOrder})

        for _, option in ipairs(options) do
            local optBtn = newInstance("TextButton", {
                Size = UDim2.new(1, 0, 0, optH),
                BackgroundColor3 = COLOR.panel,
                Text = option.name, TextColor3 = COLOR.text,
                Font = currentFontPreset.regular, TextSize = math.max(10, math.floor(12 * textScale)), ZIndex = 51, Parent = panel,
            })
            bindFont(optBtn, "regular")
            optBtn.MouseButton1Click:Connect(function()
                onSelect(option)
                toggleButton.Text = labelPrefix .. option.name
                closeActiveOverlay()
            end)
        end
        activeOverlay = panel
    end)
end

-- Color picker: opens a compact grid of swatches with a Light/Darken section
-- (shade the pending color) and a Save button that commits it to the color
-- system -- clicking a swatch only marks it, Save commits it.
local pickerPanel = nil
local function closePickers()
    if pickerPanel and pickerPanel.Parent then pickerPanel:Destroy() end
    pickerPanel = nil
end

local function createColorPicker(button, onPick, initial)
    button.MouseButton1Click:Connect(function()
        if pickerPanel and pickerPanel.Parent then
            closePickers()
            return
        end
        closePickers()

        -- Grid: capped at 4 visible rows; if the color table ever outgrows that
        -- the swatches scroll instead of pushing the panel off-screen.
        local cols, cellH, maxRows = 4, 24, 4
        local rows = math.ceil(#ColorPalette / cols)
        local visibleRows = math.min(rows, maxRows)
        local pad, gap = 6, 4
        local gridCanvasH = pad * 2 + rows * cellH + (rows - 1) * gap
        local gridH = pad * 2 + visibleRows * cellH + (visibleRows - 1) * gap
        local footH = 64
        local panelH = gridH + footH

        local pending = (initial and initial()) or nil
        -- Light/dark lock: shading is computed relative to the ORIGINAL
        -- (unshaded) color and the factor is clamped to a small band, so a few
        -- presses can never wash the color to white/black.
        local baseColor = pending
        local shadeFactor = 0

        local pos, size = button.AbsolutePosition, button.AbsoluteSize
        local cam = workspace.CurrentCamera
        local vh = (cam and cam.ViewportSize.Y) or 600
        local vw = (cam and cam.ViewportSize.X) or 800

        -- Clamp so the table never slides off the right edge on narrow windows.
        local panelW = math.min(236, vw - 8)
        local cellW = (panelW - pad * 2 - gap * (cols - 1)) / cols
        local px = math.max(2, math.min(pos.X + (size.X - panelW) / 2, vw - panelW - 2))
        local openUp = (pos.Y - panelH - 4) < 0
        local y = openUp and math.max(2, pos.Y + size.Y + 4) or math.max(2, pos.Y - panelH - 4)
        y = math.min(y, vh - panelH - 4)

        -- Same stacking setup as the working dropdowns: panel at ZIndex 51 with
        -- every child at ZIndex 51 too, so the swatch grid can never render
        -- behind the panel itself or the window shell.
        local panel = newInstance("Frame", {
            Name = "PickerPanel",
            Size = UDim2.new(0, panelW, 0, panelH),
            Position = UDim2.new(0, px, 0, y),
            BackgroundColor3 = COLOR.panel, ZIndex = 51, Parent = HUB.overlayLayer,
        })
        round(panel, 6)
        stroke(panel, COLOR.border, 1)
        pickerPanel = panel

        local gridHost = newInstance("ScrollingFrame", {
            Size = UDim2.new(1, 0, 0, gridH),
            Position = UDim2.new(0, 0, 0, 0),
            BackgroundTransparency = 1, ScrollBarThickness = 4,
            CanvasSize = UDim2.new(0, 0, 0, gridCanvasH),
            ZIndex = 51, Parent = panel,
        })

        local selectedEdge = nil
        local selectedColor = nil
        local selectedCell = nil
        local function selectCell(cell, edge, color)
            if selectedEdge and selectedColor then
                selectedEdge.Color = selectedColor
                selectedEdge.Thickness = 1
            end
            pending = color
            baseColor = color
            shadeFactor = 0
            selectedCell = cell
            selectedEdge = edge
            selectedColor = color
            if cell then cell.BackgroundColor3 = color end
            if edge then
                edge.Color = Color3.fromRGB(255, 255, 255)
                edge.Thickness = 2
            end
        end
        for i, o in ipairs(ColorPalette) do
            local cellColor = o.value
            local cell = newInstance("TextButton", {
                Size = UDim2.new(0, cellW, 0, cellH),
                Position = UDim2.new(0, pad + (i - 1) % cols * (cellW + gap), 0, pad + math.floor((i - 1) / cols) * (cellH + gap)),
                BackgroundColor3 = cellColor,
                Text = o.name,
                TextColor3 = (cellColor.R + cellColor.G + cellColor.B) > 1.8 and Color3.fromRGB(20, 20, 20) or Color3.fromRGB(255, 255, 255),
                Font = currentFontPreset.bold, TextSize = 10, ZIndex = 51, Parent = gridHost,
            })
            round(cell, 4)
            local edge = newInstance("UIStroke", {Color = cellColor, Thickness = 1, Parent = cell})
            if cellColor == pending then
                selectCell(cell, edge, cellColor)
            end
            cell.MouseButton1Click:Connect(function()
                selectCell(cell, edge, cellColor)
            end)
        end

        -- Light/dark section: a standalone shade group. Each press applies a
        -- FIXED step (0.15) from the original unshaded color and the factor is
        -- clamped to +/-0.45, so a few presses can't wash a slate color to
        -- white or black. Save still commits the shaded value to the theme.
        local lightDarkLabel = newInstance("TextLabel", {
            Size = UDim2.new(1, -10, 0, 12), Position = UDim2.new(0, 5, 0, gridH + 2),
            BackgroundTransparency = 1, Text = "LIGHT / DARKEN", TextXAlignment = Enum.TextXAlignment.Center,
            TextColor3 = COLOR.textDim, Font = currentFontPreset.bold, TextSize = 10, ZIndex = 51, Parent = panel,
        })
        bindFont(lightDarkLabel, "bold")

        local SHADE_STEP, SHADE_MAX = 0.15, 0.45
        local function applyShade()
            if not pending then return end
            pending = shadeColor(baseColor, shadeFactor)
            if selectedCell then selectedCell.BackgroundColor3 = pending end
            if selectedEdge then
                selectedEdge.Color = Color3.fromRGB(255, 255, 255)
                selectedEdge.Thickness = 2
            end
            selectedColor = pending
        end

        local lightenBtn = newInstance("TextButton", {
            Size = UDim2.new(0.5, -5, 0, 22), Position = UDim2.new(0, 5, 0, gridH + 16),
            BackgroundColor3 = COLOR.playerBtn, Text = "Lighten", TextColor3 = Color3.fromRGB(255, 255, 255),
            Font = currentFontPreset.bold, TextSize = 11, ZIndex = 51, Parent = panel,
        })
        round(lightenBtn, 5)
        stroke(lightenBtn, COLOR.border, 1)
        bindFont(lightenBtn, "bold")
        lightenBtn.MouseButton1Click:Connect(function()
            shadeFactor = math.min(shadeFactor + SHADE_STEP, SHADE_MAX)
            applyShade()
        end)

        local darkenBtn = newInstance("TextButton", {
            Size = UDim2.new(0.5, -5, 0, 22), Position = UDim2.new(0.5, 0, 0, gridH + 16),
            BackgroundColor3 = COLOR.playerBtn, Text = "Darken", TextColor3 = Color3.fromRGB(255, 255, 255),
            Font = currentFontPreset.bold, TextSize = 11, ZIndex = 51, Parent = panel,
        })
        round(darkenBtn, 5)
        stroke(darkenBtn, COLOR.border, 1)
        bindFont(darkenBtn, "bold")
        darkenBtn.MouseButton1Click:Connect(function()
            shadeFactor = math.max(shadeFactor - SHADE_STEP, -SHADE_MAX)
            applyShade()
        end)

        local saveBtn = newInstance("TextButton", {
            Size = UDim2.new(0.5, -5, 0, 22), Position = UDim2.new(0, 5, 0, gridH + 42),
            BackgroundColor3 = COLOR.accent2, Text = "Save", TextColor3 = Color3.fromRGB(255, 255, 255),
            Font = currentFontPreset.bold, TextSize = 11, ZIndex = 51, Parent = panel,
        })
        round(saveBtn, 5)
        bindFont(saveBtn, "bold")
        saveBtn.MouseButton1Click:Connect(function()
            if pending then
                local ok = pcall(onPick, pending)
                if not ok then showToast("Save failed", "warn") end
            end
            closePickers()
        end)

        local cancelBtn = newInstance("TextButton", {
            Size = UDim2.new(0.5, -5, 0, 22), Position = UDim2.new(0.5, 0, 0, gridH + 42),
            BackgroundColor3 = COLOR.panel, Text = "Cancel", TextColor3 = COLOR.textDim,
            Font = currentFontPreset.bold, TextSize = 11, ZIndex = 51, Parent = panel,
        })
        round(cancelBtn, 5)
        stroke(cancelBtn, COLOR.border, 1)
        bindFont(cancelBtn, "bold")
        cancelBtn.MouseButton1Click:Connect(function()
            closePickers()
        end)
    end)
end

-- ============================================================
-- TOGGLE SWITCH COMPONENT
-- ============================================================
local function createToggleSwitch(parent, position, style)
    -- Two styles: "check" renders a visible clickable button (highlight on
    -- hover, checkmark when ON, blank when OFF); default is the sliding pill.
    local isCheck = (style == "check")

    local track = newInstance("TextButton", {
        Size = isCheck and UDim2.new(0, 22, 0, 20) or UDim2.new(0, 40, 0, 20),
        Position = position,
        BackgroundColor3 = COLOR.switchOff,
        BorderSizePixel = 0,
        Text = "", AutoButtonColor = false, Parent = parent,
    })

    local state = false
    local callback = nil
    local knob = nil
    local mark = nil

    local function restColor()
        return state and COLOR.accent or COLOR.switchOff
    end

    if isCheck then
        round(track, 5)
        stroke(track, COLOR.border, 1)
        mark = newInstance("TextLabel", {
            Size = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            Text = "",
            TextColor3 = Color3.fromRGB(235, 235, 235),
            TextXAlignment = Enum.TextXAlignment.Center,
            TextYAlignment = Enum.TextYAlignment.Center,
            Font = currentFontPreset.bold,
            TextSize = 15,
            Parent = track,
        })
        bindFont(mark, "bold")
        track.MouseEnter:Connect(function()
            track.BackgroundColor3 = shadeColor(restColor(), -0.16)
        end)
        track.MouseLeave:Connect(function()
            track.BackgroundColor3 = restColor()
        end)
    else
        round(track, 10)
        knob = newInstance("Frame", {
            Size = UDim2.new(0, 16, 0, 16),
            Position = UDim2.new(0, 2, 0.5, -8),
            BackgroundColor3 = Color3.fromRGB(235, 235, 235),
            Parent = track,
        })
        round(knob, 8)
    end

    table.insert(switchTracks, {track = track, mark = mark, get = function() return state end})

    local function setState(newState, silent)
        state = newState
        if isCheck then
            mark.Text = newState and "\xE2\x9C\x93" or ""
            track.BackgroundColor3 = restColor()
        else
            TweenService:Create(track, TweenInfo.new(0.12), {
                BackgroundColor3 = newState and COLOR.accent or COLOR.switchOff
            }):Play()
            TweenService:Create(knob, TweenInfo.new(0.12), {
                Position = newState and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
            }):Play()
        end
        if not silent and callback then callback(newState) end
    end

    local function setCallback(fn) callback = fn end

    track.MouseButton1Click:Connect(function() setState(not state) end)

    return track, setState, function() return state end, setCallback
end

-- ============================================================
-- SECTION BUILDER (labeled toggle row used across every page)
-- id (optional, V4): dotted path like "server.teleport" -- the
-- row registers itself as HUB.rows.server.teleport = { status,
-- set, get, cb }, so wiring code finds rows by name instead of
-- juggling return values. Without an id the legacy 4-return
-- signature is unchanged.
-- ============================================================
local function createSwitchRow(parent, y, headerText, keyHint, id, style)
    local header = newInstance("TextLabel", {
        Size = UDim2.new(0.55, 0, 0, 16), Position = UDim2.new(0, 5, 0, y),
        BackgroundTransparency = 1, Text = headerText .. (keyHint and ("  (" .. keyHint .. ")") or ""),
        TextXAlignment = Enum.TextXAlignment.Left, TextColor3 = COLOR.textDim,
        Font = currentFontPreset.bold, TextSize = 11, Parent = parent,
    })
    bindFont(header, "bold")

    local status = newInstance("TextLabel", {
        Size = UDim2.new(0.25, 0, 0, 16), Position = UDim2.new(0.55, 0, 0, y),
        BackgroundTransparency = 1, Text = "Off", TextXAlignment = Enum.TextXAlignment.Right,
        TextColor3 = COLOR.off, Font = currentFontPreset.bold, TextSize = 11, Parent = parent,
    })
    bindFont(status, "bold")

    local togglePos = (style == "check") and UDim2.new(1, -27, 0, y - 2) or UDim2.new(1, -45, 0, y - 2)
    local track, setSwitch, getSwitch, setSwitchCallback = createToggleSwitch(parent, togglePos, style)
    if id then
        local parts = {}
        for part in string.gmatch(id, "[^%.]+") do parts[#parts + 1] = part end
        local node = HUB.rows
        for i = 1, #parts - 1 do
            node[parts[i]] = node[parts[i]] or {}
            node = node[parts[i]]
        end
        node[parts[#parts]] = { status = status, set = setSwitch, get = getSwitch, cb = setSwitchCallback }
    end
    return status, setSwitch, getSwitch, setSwitchCallback
end

-- V6: per-tab state (active refresh fn + page id) kept in a plain table;
-- Roblox instances reject custom members, so nothing hangs off the buttons.
local tabStates = {}
-- [V6 FIX] Forward declaration: Luau locals are only in scope AFTER their
-- declaration. The tab click closures below reference showPage, which is
-- defined much later (Part E) -- without this forward decl the closures
-- would bind the GLOBAL showPage (nil), killing every tab switch and any
-- button that routes through showPage at runtime.
local showPage

local function makeTab(text, index, pageId)
    local count = 4
    local btn = newInstance("TextButton", {
        Size = UDim2.new(1/count, -3, 1, 0), Position = UDim2.new((index - 1)/count, index - 1, 0, 0),
        BackgroundColor3 = COLOR.panel, Text = text, TextColor3 = COLOR.text,
        Font = currentFontPreset.bold, TextSize = 11, Parent = HUB.tabBar,
    })
    round(btn, 6)
    bindFont(btn, "bold")
    -- V6 design: hover brightens the tab; the active tab gets an
    -- accent2 underline strip instead of a full background change.
    local underline = newInstance("Frame", {
        Size = UDim2.new(1, -10, 0, 2), Position = UDim2.new(0.5, 0, 1, -3),
        AnchorPoint = Vector2.new(0.5, 1), BackgroundColor3 = COLOR.accent2,
        BorderSizePixel = 0, Visible = false, Parent = btn,
    })
    bindThemed(underline, "BackgroundColor3", "accent2")
    local function refreshTabState()
        local active = HUB.currentPage == pageId
        underline.Visible = active
        btn.BackgroundColor3 = active and COLOR.accent2 or COLOR.panel
        btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or COLOR.text
    end
    btn.MouseEnter:Connect(function()
        if HUB.currentPage ~= pageId then btn.BackgroundColor3 = shadeColor(COLOR.panel, -0.12) end
    end)
    btn.MouseLeave:Connect(function()
        btn.BackgroundColor3 = (HUB.currentPage == pageId) and COLOR.accent2 or COLOR.panel
    end)
    btn.MouseButton1Click:Connect(function()
        showPage(pageId)
        refreshTabState()
    end)
    -- [V6 FIX] Instances reject custom fields ("X is not a valid member
    -- of TextButton"), so tab bookkeeping lives in a script-side table
    -- keyed by the button instead of on the button itself.
    tabStates[btn] = { refresh = refreshTabState, pageId = pageId }
    return btn
end

local UiPresetSlots = {}
for i, imp in ipairs(IMPORTED_UI_PRESETS) do
    if type(imp) == "table" then UiPresetSlots[i] = imp end
end

local function uiPresetCapture()
    return {
        name = currentPreset.name,
        colors = {
            accent = COLOR.accent, accent2 = COLOR.accent2, accent3 = COLOR.accent3,
            bg = COLOR.bg, text = COLOR.text,
        },
        size = currentSizePreset.name,
        font = currentFontPreset.name,
        border = currentBorderPreset.name,
        edging = currentEdgingFilter and currentEdgingFilter.name or "Default",
    }
end

local function uiPresetApply(slot, silent)
    if not slot then return end
    if slot.colors and slot.colors.accent then
        applyTheme({
            name = slot.name or "Custom",
            accent = slot.colors.accent,
            accent2 = slot.colors.accent2 or COLOR.accent2,
            accent3 = slot.colors.accent3 or COLOR.accent3,
            bg = slot.colors.bg,
            text = slot.colors.text,
        })
    end
    for _, p in ipairs(SIZE_PRESETS) do
        if p.name == slot.size then applySize(p) break end
    end
    for _, p in ipairs(FONT_PRESETS) do
        if p.name == slot.font then applyFont(p) break end
    end
    for _, p in ipairs(BORDER_EDGING_PRESETS) do
        if p.name == slot.border then applyBorderStyle(p) applyCornerStyle(p) break end
    end
    if HUB.applyEdgingFilterByName then HUB.applyEdgingFilterByName(slot.edging) end
    if not silent then showToast("UI preset: " .. (slot.name or "?"), "on") end
end

local function uiPresetSave(i)
    local cap = uiPresetCapture()
    cap.name = currentPreset.name .. " " .. tostring(i)
    UiPresetSlots[i] = cap
    _G.schizhubUiPresets = UiPresetSlots
    _G.schizhubLastUiSlot = i
    return cap
end

-- Emits the Lua snippet for the slot; paste it into IMPORTED_UI_PRESETS
-- at the top of this script to make the look permanent.
local function uiPresetCopy(i)
    local slot = UiPresetSlots[i]
    if not slot then return false end
    local c = slot.colors or {}
    local lines = {
        "IMPORTED_UI_PRESETS[" .. tostring(i) .. "] = {",
        "    name = " .. string.format("%q", slot.name or "Custom") .. ",",
        "    colors = {",
        "        accent = Color3.fromRGB(" .. PaletteRgb(c.accent or COLOR.accent) .. "),",
        "        accent2 = Color3.fromRGB(" .. PaletteRgb(c.accent2 or COLOR.accent2) .. "),",
        "        accent3 = Color3.fromRGB(" .. PaletteRgb(c.accent3 or COLOR.accent3) .. "),",
        "        bg = Color3.fromRGB(" .. PaletteRgb(c.bg or COLOR.bg) .. "),",
        "        text = Color3.fromRGB(" .. PaletteRgb(c.text or COLOR.text) .. "),",
        "    },",
        "    size = " .. string.format("%q", slot.size or "Medium") .. ",",
        "    font = " .. string.format("%q", slot.font or "Gotham") .. ",",
        "    border = " .. string.format("%q", slot.border or "Default") .. ",",
        "    edging = " .. string.format("%q", slot.edging or "Default") .. ",",
        "},",
    }
    local ok = pcall(setclipboard, table.concat(lines, "\n"))
    return ok
end

-- The whole page build lives in its own function so the main chunk stays under
-- Luau's 200-local-register cap as the settings sections keep growing.
local function buildSettingsPage()
-- Edging color filters: derives a border color from the current mode and
-- pushes it into the theme system so every themed edge updates at once.
-- Background is preselected (looked up by mode, not array index, since the
-- Background button sits on its own row above the centered trio).
local edgingFilterBtns = {}
-- [FIX 1] lighten/darken derive from the live theme border (COLOR.border)
-- instead of the hardcoded defaultBorder, and darken uses a 0.47 multiplier.
local function edgingFilterColor(filter)
    if filter.mode == "accent" then return COLOR.accent end
    if filter.mode == "accent2" then return COLOR.accent2 end
    if filter.mode == "bg" then return COLOR.bg end
    if filter.mode == "lighten" then
        local c = COLOR.border
        local l = 0.35
        return Color3.fromRGB(
            math.floor(c.R * 255 + (255 - c.R * 255) * l),
            math.floor(c.G * 255 + (255 - c.G * 255) * l),
            math.floor(c.B * 255 + (255 - c.B * 255) * l)
        )
    end
    if filter.mode == "darken" then
        local c = COLOR.border
        return Color3.fromRGB(math.floor(c.R * 255 * 0.47), math.floor(c.G * 255 * 0.47), math.floor(c.B * 255 * 0.47))
    end
    return defaultBorder
end
local function updateEdgingFilterLabels()
    for _, e in ipairs(edgingFilterBtns) do
        local active = (e.filter == currentEdgingFilter)
        e.btn.BackgroundColor3 = active and COLOR.accent or COLOR.playerBtn
        e.btn.TextColor3 = active and Color3.fromRGB(30, 30, 30) or COLOR.text
    end
end
local function applyEdgingFilter(filter)
    currentEdgingFilter = filter
    COLOR.border = edgingFilterColor(filter)
    refreshTheme()
    updateEdgingFilterLabels()
end

do
    local SpdView = HUB.views.spd
    local SPD_SLIDER_MIN, SPD_SLIDER_MAX = 15, 100

    -- KACHOW MODE state (shared with the :kachow command in the command engine)
    local kachowModeOn = false
    local kachowSpeed = 50
    local kachowBaseWalk = 16
    local kachowOverride = false
    local kachowBoosting = false

    local function kachowHumanoid()
        local char = LocalPlayer.Character
        return char and char:FindFirstChildOfClass("Humanoid")
    end
    local function kachowApplySpeed(v)
        local hum = kachowHumanoid()
        if hum then hum.WalkSpeed = v end
    end
    local function kachowRestore()
        kachowApplySpeed(kachowBaseWalk)
    end

    -- The speed section (header + slider + presets) is always visible; only the
    -- Kachow branding (header / button / context) is hidden until the kachow command.
    local SpdSpeedSection = newInstance("Frame", {
        Size = UDim2.new(1, -10, 0, 54), Position = UDim2.new(0, 5, 0, 4),
        BackgroundTransparency = 1, Parent = SpdView,
    })

    newInstance("TextLabel", {

        BackgroundTransparency = 1, Text = "SPEED (" .. SPD_SLIDER_MIN .. "-" .. SPD_SLIDER_MAX .. ")", TextXAlignment = Enum.TextXAlignment.Center,
        TextColor3 = COLOR.textDim, Font = currentFontPreset.bold, TextSize = 11, Parent = SpdSpeedSection,
    })

    local SpdKachowHeader = newInstance("TextLabel", {
        Size = UDim2.new(1, -10, 0, 18), Position = UDim2.new(0, 5, 0, 66),
        BackgroundTransparency = 1, Text = "KACHOW", TextXAlignment = Enum.TextXAlignment.Center,
        TextColor3 = COLOR.accent, Font = currentFontPreset.bold, TextSize = 15, Visible = false, Parent = SpdView,
    })

    local SpdSliderTrack, SpdFill, SpdKnob, SpdValue
    local function spdApplyValue(v, live)
        v = math.clamp(math.round(v), SPD_SLIDER_MIN, SPD_SLIDER_MAX)
        kachowSpeed = v
        local frac = (v - SPD_SLIDER_MIN) / (SPD_SLIDER_MAX - SPD_SLIDER_MIN)
        if SpdKnob then SpdKnob.Position = UDim2.new(frac, -8, 0.5, -8) end
        if SpdFill then SpdFill.Size = UDim2.new(frac, 0, 1, 0) end
        if SpdValue then SpdValue.Text = tostring(v) end
        if live then
            kachowApplySpeed(kachowSpeed)
            if kachowModeOn and not kachowBoosting then
                kachowBaseWalk = kachowSpeed
            end
        end
    end

    local function spdSetSpeed(v)
        spdApplyValue(v, true)
    end

    SpdSliderTrack = newInstance("Frame", {
        Size = UDim2.new(1, -10, 0, 8), Position = UDim2.new(0, 5, 0, 16),
        BackgroundColor3 = COLOR.panel, Parent = SpdSpeedSection,

        TextColor3 = COLOR.text, Font = currentFontPreset.bold, TextSize = 12, Parent = SpdSpeedSection,
    })
    spdApplyValue(kachowSpeed, false)

    local spdDragging = false
    local function spdUpdateFromMouse()
        local abs = SpdSliderTrack.AbsolutePosition
        local w = SpdSliderTrack.AbsoluteSize.X
        if w <= 0 then return end
        local frac = (UserInputService:GetMouseLocation().X - abs.X) / w
        spdApplyValue(SPD_SLIDER_MIN + (SPD_SLIDER_MAX - SPD_SLIDER_MIN) * math.clamp(frac, 0, 1), true)
    end
    SpdSliderTrack.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            spdUpdateFromMouse()
            spdDragging = true
        end
    end)
    SpdKnob.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            spdDragging = true
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if spdDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            spdUpdateFromMouse()
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then spdDragging = false end
    end)

    for i, opt in ipairs(SPEED_OPTIONS) do
        local b = newInstance("TextButton", {

    local function setKachowMode(on)
        if kachowModeOn == on then return end
        kachowModeOn = on
        if on then
            kachowReveal()
            local hum = kachowHumanoid()
            kachowBaseWalk = hum and hum.WalkSpeed or 16
            if kachowOverride then kachowApplySpeed(kachowSpeed) end
        else
            kachowOverride = false
            kachowBoosting = false
            kachowRestore()
        end
        SpdModeBtn.Text = on and "KACHOW MODE: ON" or "KACHOW MODE: OFF"
        showToast(on and "Kachow mode on" or "Kachow mode off", on and "on" or "off")
    end
    SpdModeBtn.MouseButton1Click:Connect(function()
        setKachowMode(not kachowModeOn)
    end)

    -- G: hold = smooth ramp from base speed (16) up to max (100) over 6.5s at
    -- an even pace (boost, only while Kachow mode is on). H = toggle speed on/off.
    local spdKeyHeld = false
    local spdRampTask = nil
    local SPD_RAMP_DURATION = 6.5

    local function spdStopRamp()
        if spdRampTask then
            spdRampTask:Cancel()
            spdRampTask = nil
        end
    end

    local function spdStartRamp()
        spdStopRamp()
        local startSpeed = kachowBaseWalk
        local targetSpeed = SPD_SLIDER_MAX
        local startTime = os.clock()
        spdRampTask = task.spawn(function()
            while true do
                local t = (os.clock() - startTime) / SPD_RAMP_DURATION
                if t >= 1 then
                    kachowApplySpeed(targetSpeed)
                    break
                end
                kachowApplySpeed(startSpeed + (targetSpeed - startSpeed) * t)
                task.wait(0.05)
            end
            spdRampTask = nil
        end)
    end

    UserInputService.InputBegan:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.G and kachowModeOn then
            spdKeyHeld = true
            kachowBoosting = true
            spdStartRamp()
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.G and kachowModeOn and spdKeyHeld then
            spdKeyHeld = false
            kachowBoosting = false
            spdStopRamp()
            if not kachowOverride then kachowRestore() end
        end
    end)
    UserInputService.InputBegan:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.H then
            kachowOverride = not kachowOverride
            if kachowOverride then
                kachowApplySpeed(kachowSpeed)
                showToast("Speed: ON", "on")
            else
                kachowRestore()
                showToast("Speed: OFF", "off")
            end
        end
    end)

    LocalPlayer.CharacterAdded:Connect(function()
        if kachowModeOn and kachowOverride then
            task.wait(0.2)
            kachowApplySpeed(kachowSpeed)
        end
    end)

    kachowApi = {
        setMode = setKachowMode,
        isOn = function() return kachowModeOn end,
    }
end

local function EssTeamColor(plr)
    if plr == LocalPlayer then return EssColors.Ally end
    -- Team-smart: same team as the local player = Ally, otherwise Enemy.
    -- Works for 2-team games (Red vs Blue, Killer vs Survivor); falls
    -- back to everyone-else-is-Enemy when no teams are set.
    local myTeam = LocalPlayer.Team
    local theirTeam = plr.Team
    if myTeam ~= nil and theirTeam ~= nil then
        return (myTeam == theirTeam) and EssColors.Ally or EssColors.Enemy
    end
    return EssColors.Enemy
end

local function EssFindESP(obj)
    for _, v in ipairs(EssItemESP:GetChildren()) do
        if v:IsA("ObjectValue") and v.Value == obj then return v.Parent end
    end
    return nil
end

local function EssRemoveESP(obj)
    if obj == nil then return end
    local isPlr = obj:IsA("Player")
    local folder = isPlr and EssPlayerESP or EssItemESP
    local found = isPlr and folder:FindFirstChild("ESP Crap_" .. obj.Name) or EssFindESP(obj)
    if found then found:Destroy() end
end

local function EssCreateESP(obj)
    if obj == nil then return end
    local isPlr = obj:IsA("Player")
    local folder = isPlr and EssPlayerESP or EssItemESP
    local subject = isPlr and obj.Character or obj
    if folder:FindFirstChild("ESP Crap_" .. obj.Name) then return end
    if subject == nil then return end
    local head = subject:FindFirstChild("Head")
    if head == nil then return end

    local bb = newInstance("BillboardGui", {
        Adornee = head, ExtentsOffset = Vector3.new(0, 1, 0),
        AlwaysOnTop = true, Size = UDim2.new(0, 5, 0, 5),
        StudsOffset = Vector3.new(0, 3, 0),
        Name = "ESP Crap_" .. obj.Name, Parent = folder,
    })
    local frame = newInstance("Frame", { Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Parent = bb })
    local col = EssTeamColor(obj)

    local nameLabel = newInstance("TextLabel", {
        Name = "Names", Size = UDim2.new(1, 0, 10, 0), Position = UDim2.new(0, 0, 0, -45),
        BackgroundTransparency = 1, Text = obj.Name, TextColor3 = col,
        Font = Enum.Font.SourceSansBold, TextSize = 13, TextStrokeTransparency = 0.5, Parent = frame,
    })
    if isPlr then
        newInstance("TextLabel", {
            Name = "Dist", Size = UDim2.new(1, 0, 10, 0), Position = UDim2.new(0, 0, 0, -35),
            BackgroundTransparency = 1, Text = "", TextColor3 = col,
            Font = Enum.Font.SourceSansBold, TextSize = 13, TextStrokeTransparency = 0.5, Parent = frame,
        })
        newInstance("TextLabel", {
            Name = "Health", Size = UDim2.new(1, 0, 10, 0), Position = UDim2.new(0, 0, 0, -25),
            BackgroundTransparency = 1, Text = "", TextColor3 = col,
            Font = Enum.Font.SourceSansBold, TextSize = 13, TextStrokeTransparency = 0.5, Parent = frame,
        })
    else
        newInstance("ObjectValue", { Value = obj, Parent = bb })
    end
end

local function EssUpdateESP(plr)
    if plr == nil then return end
    local esp = EssPlayerESP:FindFirstChild("ESP Crap_" .. plr.Name)
    if not esp then return end
    local frame = esp.Frame
    local col = EssTeamColor(plr)
    frame.Names.TextColor3 = col
    frame.Dist.TextColor3 = col
    frame.Health.TextColor3 = col

    local char = plr.Character
    local myChar = EssMyChar
    if not (myChar and char) then return end
    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not (myHRP and hrp) then return end

    local dist = (myHRP.Position - hrp.Position).Magnitude
    local hidden = dist > EssConfig.ESPLength
    frame.Names.Visible = not hidden
    frame.Dist.Visible = not hidden
    frame.Health.Visible = not hidden
    frame.Dist.Text = "Distance: " .. string.format("%.0f", dist)
    frame.Health.Text = hum and ("Health: " .. string.format("%.0f", hum.Health)) or ""
end

local function EssNpcIsNpc(v)
    return v:IsA("Model") and (v:FindFirstChild("Humanoid") or v:FindFirstChild("HumanoidRootPart"))
        and not Players:GetPlayerFromCharacter(v)
end

    if model:FindFirstChild("GhostNPC") then return end
    newInstance("Highlight", {
        Name = "GhostNPC", FillColor = EssColors.NPC, OutlineColor = EssColors.NPC,
        FillTransparency = 0.5, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop, Parent = model,
    })
    table.insert(EssNPCList, model)
end

local function EssRefreshNPCs()
    if not EssConfig.NPCESPEnabled then return end
    for _, v in ipairs(workspace:GetDescendants()) do
        if EssNpcIsNpc(v) then EssApplyNPC(v) end

end

local function EssCreateTracers(plr)
    if EssTracerData[plr.Name] or not EssTracerFolder then return end
    local part = newInstance("Part", {
        Name = plr.Name, Material = Enum.Material.Neon, Transparency = 1,
        Anchored = true, Locked = true, CanCollide = false, Parent = EssTracerFolder,
    })
    local box = newInstance("BoxHandleAdornment", {
        Adornee = part, Size = Vector3.new(1, 1, 1), AlwaysOnTop = true,
        ZIndex = 5, Transparency = 0, Color3 = EssTeamColor(plr), Parent = part,
    })
    local char = plr.Character
    EssTracerData[plr.Name] = {
        Part = part, Box = box,
        HRP = char and char:FindFirstChild("HumanoidRootPart") or nil,
    }
end

local function EssRemoveTracers(plr)
    local d = EssTracerData[plr.Name]
    if d then
        d.Part:Destroy()
        EssTracerData[plr.Name] = nil
    end
end

local function EssUpdateTracer(plr, d)
    if not EssConfig.TracersEnabled then return end
    local myChar = EssMyChar
    if not myChar then return end
    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
    local hrp = d.HRP
    if not (myHRP and hrp and hrp.Parent) then return end

    local col = EssTeamColor(plr)
    local dist = (myHRP.Position - hrp.Position).Magnitude
    local inVehicle = myChar:FindFirstChild("InVehicle") or hrp.Parent:FindFirstChild("InVehicle")
    if dist < EssConfig.TracersLength and not inVehicle and EssCam then
        local ray = EssCam:ScreenPointToRay(EssCam.ViewportSize.X / 2, EssCam.ViewportSize.Y, 0)
        local len = (ray.Origin - (hrp.Position - Vector3.new(0, 3, 0))).Magnitude
        d.Part.Transparency = 1
        d.Part.Size = Vector3.new(0.05, 0.05, len)
        d.Part.CFrame = CFrame.new(ray.Origin, (hrp.Position - Vector3.new(0, 4.5, 0))) * CFrame.new(0, 0, -len / 2)
        d.Part.BrickColor = BrickColor.new(col)
        d.Box.Transparency = 0
        d.Box.Size = Vector3.new(0.001, 0.001, len)
        d.Box.Color3 = col
    else
        d.Part.Transparency = 1
        d.Box.Transparency = 1
    end
end

    local lastFire = 0

    -- Line-of-sight: raycast from the camera; a hit before the target
    -- means a wall is in the way (skip the target).
    local function aimVisible(targetPos)
        local camPos = EssCam.CFrame.Position
        local dir = targetPos - camPos
        local dist = dir.Magnitude
        if dist < 0.01 then return true end
        local params = RaycastParams.new()
        params.FilterDescendantsInstances = { LocalPlayer.Character }
        params.FilterType = Enum.RaycastFilterType.Blacklist
        local hit = workspace:Raycast(camPos, dir.Unit * dist, params)
        return hit == nil or (hit.Position - camPos).Magnitude >= dist - 1
    end

    -- Cycle mode: Tab switches the locked aim target (no teleport).
    local function cycleAimTarget()
        if not HUB.cycleMode then return end
        local list = {}
        local myTeam = LocalPlayer.Team
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local theirTeam = plr.Team
                if not (myTeam ~= nil and theirTeam ~= nil and myTeam == theirTeam) then
                    local char = plr.Character
                    local part = (char and char:FindFirstChild("Head")) or (char and char:FindFirstChild("HumanoidRootPart"))
                    if part and aimVisible(part.Position) then table.insert(list, plr) end
                end
            end
        end
        if #list == 0 then showToast("No visible target", "warn") return end
        local idx = 0
        if HUB.aimLockTarget then
            for i, plr in ipairs(list) do
                if plr == HUB.aimLockTarget then idx = i break end
            end
        end
        HUB.aimLockTarget = list[idx % #list + 1]
        showToast("Aim target: " .. HUB.aimLockTarget.Name, "on")
    end
    HUB.cycleAimTarget = cycleAimTarget

    -- Auto-aim toward the locked target (or nearest visible on-screen
    -- target). Shooter Farm forces aim + auto-fires.
    RunService:BindToRenderStep("PSE_Aim_Update", Enum.RenderPriority.Character.Value, function()
        local aimActive = EssConfig.AimAssistEnabled or EssConfig.ShooterFarmEnabled
        if EssConfig.HoldToAim and not EssConfig.ShooterFarmEnabled then
            aimActive = aimActive and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
        end
        if not aimActive then return end
        local cam = EssCam
        local viewport = cam.ViewportSize
        local centerX, centerY = viewport.X / 2, viewport.Y / 2
        local camPos = cam.CFrame.Position
        local myTeam = LocalPlayer.Team

        local function aimPart(plr)
            local char = plr.Character
            local head = char and char:FindFirstChild("Head")
            return head or (char and char:FindFirstChild("HumanoidRootPart"))
        end

        -- Prefer the locked target while it is still valid + visible.
        local target, targetPart = nil, nil
        if HUB.aimLockTarget and HUB.aimLockTarget.Parent then
            local p = aimPart(HUB.aimLockTarget)
            if p and aimVisible(p.Position) then
                target, targetPart = HUB.aimLockTarget, p
            end
        end
        if not target then
            local bestDist = EssConfig.AimRadius
            local bestHealth = math.huge
            local lowestHP = (EssConfig.AimPriority == "lowest")
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer then
                    local theirTeam = plr.Team
                    if not (myTeam ~= nil and theirTeam ~= nil and myTeam == theirTeam) then
                        local p = aimPart(plr)
                        if p and aimVisible(p.Position) then
                            local vp = cam:WorldToViewportPoint(p.Position)
                            if vp and vp.Z > 0 then
                                local dx, dy = vp.X - centerX, vp.Y - centerY
                                local dist = math.sqrt(dx * dx + dy * dy)
                                if dist <= EssConfig.AimRadius then
                                    if lowestHP then
                                        local hum = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid")
                                        local hp = hum and hum.Health or math.huge
                                        if hp < bestHealth then bestHealth = hp; target, targetPart = plr, p end
                                    elseif dist < bestDist then
                                        bestDist = dist; target, targetPart = plr, p
                                    end
                                end
                            end
                        end
                    end
                end
            end
            if target and EssConfig.AimAssistEnabled then HUB.aimLockTarget = target end
        end
        if not target or not targetPart then return end

        local targetDir = (targetPart.Position - camPos).Unit
        local strength = math.clamp(EssConfig.AimStrength, 0, 1)
        local newDir = cam.CFrame.LookVector:Lerp(targetDir, strength).Unit
        cam.CFrame = CFrame.lookAt(camPos, camPos + newDir)

        if EssConfig.ShooterFarmEnabled then
            local now = tick()
            if now - lastFire >= EssConfig.FireRate then
                lastFire = now
                local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                if tool then pcall(function() tool:Activate() end) end
            end
        end
    end)
end

                        f.Size = UDim2.new(0, width, 0, height)
                        f.Visible = true
                    end
                end
            end
        end
    end)
end

local cmds = {}
local function addcmd(name, alias, func, hidden)
    table.insert(cmds, {NAME = name, ALIAS = alias or {}, FUNC = func, HIDDEN = hidden})
end

local function findCmd(name)
    for _, c in ipairs(cmds) do
        if c.NAME == name then return c end
        for _, a in ipairs(c.ALIAS) do
            if a == name then return c end
        end
    end
end

local function notify(text, text2)
    showToast(text .. (text2 and (" - " .. text2) or ""), "textDim")
end

local function getRoot(char)
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
end

local function isNumber(str)
    return tonumber(str) ~= nil
end

local function splitString(str, delim)
    local broken = {}
    for w in string.gmatch(str, "[^" .. delim .. "]+") do
        table.insert(broken, w)
    end
    return broken
end

local function getPlayer(list, speaker)
    if not list then return {speaker} end
    local found = {}
    for _, name in ipairs(splitString(list, ",")) do
        name = name:gsub("^[+-]", "")
        local lower = name:lower()
        if lower == "all" then
            for _, p in ipairs(Players:GetPlayers()) do table.insert(found, p) end
        elseif lower == "others" then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= speaker then table.insert(found, p) end
            end
        elseif lower == "me" then
            table.insert(found, speaker)
        elseif lower == "random" then
            local others = {}
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= speaker then table.insert(others, p) end
            end
            if #others > 0 then table.insert(found, others[math.random(1, #others)]) end
        else
            local isDisplay = name:sub(1, 1) == "@"
            local search = isDisplay and name:sub(2) or name
            if search == "" then continue end
            for _, p in ipairs(Players:GetPlayers()) do
                local match = (p.Name:lower():sub(1, #search) == search:lower())
                    or (p.DisplayName:lower():sub(1, #search) == search:lower())
                if match then table.insert(found, p) end
            end
        end
    end
    return found
end

local function execCmd(cmdStr, speaker)
    cmdStr = cmdStr:gsub("%s+$", "")
    cmdStr = cmdStr:gsub("^%s*", "")
    cmdStr = cmdStr:gsub("^;", "")
    cmdStr = cmdStr:gsub("^/e ", "")
    task.spawn(function()
        local commandsToRun = splitString(cmdStr, "\\")
        for _, v in ipairs(commandsToRun) do
            local x, y, num = v:find("^(%d+)%^")
            local cmdDelay = 0
            local infTimes = false
            if num then
                v = v:sub(y + 1)
                local x, y, del = v:find("^([%d%.]+)%^")
                if del then
                    v = v:sub(y + 1)
                    cmdDelay = tonumber(del) or 0
                end
            else
                local x = v:find("^inf%^")
                if x then
                    infTimes = true
                    v = v:sub(x + 4)
                    local x, y, del = v:find("^([%d%.]+)%^")
                    if del then
                        v = v:sub(y + 1)
                        cmdDelay = (tonumber(del) or 1)
                    else
                        cmdDelay = 1
                    end
                end
            end
            num = tonumber(num or 1)

            local args = splitString(v, " ")
            -- [V6] Empty / whitespace-only commands no longer shout
            -- "Unknown command: nil" into the toast.
            if #args == 0 or args[1] == "" then continue end
            local cmd = findCmd(args[1])
            if cmd then
                table.remove(args, 1)
                if infTimes then
                    while true do
                        pcall(cmd.FUNC, args, speaker)
                        task.wait(cmdDelay)
                    end
                else
                    for rep = 1, num do
                        pcall(cmd.FUNC, args, speaker)
                        if cmdDelay ~= 0 then task.wait(cmdDelay) end
                    end
                end
            else
                showToast("Unknown command: " .. tostring(args[1]), "warn")
            end
        end
    end)
end
HUB.execCmd = execCmd

local FLYING = false

local function sFLY()
    repeat task.wait() until LocalPlayer.Character and getRoot(LocalPlayer.Character) and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    local flyMouse = LocalPlayer:GetMouse()
    if flyKeyDown or flyKeyUp then flyKeyDown:Disconnect() flyKeyUp:Disconnect() end
    local T = getRoot(LocalPlayer.Character)
    local CONTROL = {F = 0, B = 0, L = 0, R = 0, Q = 0, E = 0}
    local lCONTROL = {F = 0, B = 0, L = 0, R = 0, Q = 0, E = 0}
    local SPEED = 0
    local function FLY()
        FLYING = true
        local BG = Instance.new("BodyGyro")
        local BV = Instance.new("BodyVelocity")
        BG.P = 9e4
        BG.Parent = T
        BV.Parent = T
        BG.maxTorque = Vector3.new(9e9, 9e9, 9e9)
        BG.cframe = T.CFrame
        BV.velocity = Vector3.new(0, 0, 0)
        BV.maxForce = Vector3.new(9e9, 9e9, 9e9)
        task.spawn(function()
            repeat task.wait()
                local char = LocalPlayer.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if hum then hum.PlatformStand = true end
                if CONTROL.L + CONTROL.R ~= 0 or CONTROL.F + CONTROL.B ~= 0 or CONTROL.Q + CONTROL.E ~= 0 then
                    SPEED = 50
                elseif SPEED ~= 0 then
                    SPEED = 0
                end
                if (CONTROL.L + CONTROL.R) ~= 0 or (CONTROL.F + CONTROL.B) ~= 0 or (CONTROL.Q + CONTROL.E) ~= 0 then
                    BV.velocity = ((workspace.CurrentCamera.CoordinateFrame.lookVector * (CONTROL.F + CONTROL.B)) + ((workspace.CurrentCamera.CoordinateFrame * CFrame.new(CONTROL.L + CONTROL.R, (CONTROL.F + CONTROL.B + CONTROL.Q + CONTROL.E) * 0.2, 0).p) - workspace.CurrentCamera.CoordinateFrame.p)) * SPEED
                    lCONTROL = {F = CONTROL.F, B = CONTROL.B, L = CONTROL.L, R = CONTROL.R}
                elseif SPEED ~= 0 then
                    BV.velocity = ((workspace.CurrentCamera.CoordinateFrame.lookVector * (lCONTROL.F + lCONTROL.B)) + ((workspace.CurrentCamera.CoordinateFrame * CFrame.new(lCONTROL.L + lCONTROL.R, (lCONTROL.F + lCONTROL.B + CONTROL.Q + CONTROL.E) * 0.2, 0).p) - workspace.CurrentCamera.CoordinateFrame.p)) * SPEED
                else
                    BV.velocity = Vector3.new(0, 0, 0)
                end
                BG.cframe = workspace.CurrentCamera.CoordinateFrame
            until not FLYING
            CONTROL = {F = 0, B = 0, L = 0, R = 0, Q = 0, E = 0}
            lCONTROL = {F = 0, B = 0, L = 0, R = 0, Q = 0, E = 0}
            SPEED = 0
            BG:Destroy()
            BV:Destroy()
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.PlatformStand = false end
        end)
    end
    flyKeyDown = flyMouse.KeyDown:Connect(function(KEY)
        local k = KEY:lower()
        if k == "w" then CONTROL.F = flySpeed
        elseif k == "s" then CONTROL.B = -flySpeed
        elseif k == "a" then CONTROL.L = -flySpeed
        elseif k == "d" then CONTROL.R = flySpeed
        elseif QEfly and k == "e" then CONTROL.Q = flySpeed * 2
        elseif QEfly and k == "q" then CONTROL.E = -flySpeed * 2 end
        pcall(function() workspace.CurrentCamera.CameraType = Enum.CameraType.Track end)
    end)
    flyKeyUp = flyMouse.KeyUp:Connect(function(KEY)
        local k = KEY:lower()
        if k == "w" then CONTROL.F = 0
        elseif k == "s" then CONTROL.B = 0
        elseif k == "a" then CONTROL.L = 0
        elseif k == "d" then CONTROL.R = 0
        elseif k == "e" then CONTROL.Q = 0
        elseif k == "q" then CONTROL.E = 0 end
    end)
    FLY()
end

local fcRunning = false
local fcPart = nil
local fcConn = nil
local fcYaw, fcPitch = 0, 0

local function startFreecam()
    if fcRunning then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    fcRunning = true
    fcYaw, fcPitch = 0, 0
    fcPart = Instance.new("Part")
    fcPart.Name = "Freecam"
    fcPart.Anchored = true
    fcPart.CanCollide = false
    fcPart.Transparency = 1
    fcPart.Size = Vector3.new(1, 1, 1)
    fcPart.Parent = workspace
    fcPart.CFrame = cam.CFrame
    cam.CameraType = Enum.CameraType.Scriptable
    cam.CFrame = fcPart.CFrame
    fcConn = RunService.RenderStepped:Connect(function(dt)
        local speed = 30 * dt
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then speed = speed * 5 end
        local cf = fcPart.CFrame
        local look, right, up = cf.LookVector, cf.RightVector, cf.UpVector
        local dir = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + look end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - look end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - right end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + right end
        if UserInputService:IsKeyDown(Enum.KeyCode.E) then dir = dir + up end
        if UserInputService:IsKeyDown(Enum.KeyCode.Q) then dir = dir - up end
        if dir.Magnitude > 0 then dir = dir.Unit * speed end
        if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
            local d = UserInputService:GetMouseDelta()
            fcYaw = fcYaw - d.X * 0.003
            fcPitch = math.clamp(fcPitch - d.Y * 0.003, -1.5, 1.5)
            fcPart.CFrame = CFrame.new(fcPart.Position) * CFrame.Angles(0, fcYaw, 0) * CFrame.Angles(fcPitch, 0, 0)
        end
        fcPart.CFrame = fcPart.CFrame + dir
        cam.CFrame = fcPart.CFrame
    end)
    notify("Freecam", "Right-click to look around")
end

local function stopFreecam()
    if not fcRunning then return end
    fcRunning = false
    if fcConn then fcConn:Disconnect() fcConn = nil end
    if fcPart then fcPart:Destroy() fcPart = nil end
    local cam = workspace.CurrentCamera
    if cam then cam.CameraType = Enum.CameraType.Custom end
end

-- Fire a RemoteEvent by name/path (manual "backdoor" for server-side
-- tp-others etc.). Args are passed as strings; numbers are auto-converted.
addcmd("fire", {"firemote"}, function(args, speaker)
    local path = args[1]
    if not path then notify("Usage: :fire <remote> [args...]", nil) return end
    local remote = nil
    for _, root in ipairs({ game.ReplicatedStorage, game.Players, workspace }) do
        if root then
            for _, v in ipairs(root:GetDescendants()) do
                if v:IsA("RemoteEvent") and (v.Name == path or v:GetFullName() == path) then
                    remote = v
                    break
                end
            end
        end
        if remote then break end
    end
    if not remote then notify("Remote not found", path) return end
    local fireArgs = {}
    for i = 2, #args do
        local a = args[i]
        local num = tonumber(a)
        fireArgs[#fireArgs + 1] = num ~= nil and num or a
    end

addcmd("stats", {"getstats"}, function(args, speaker)
    local target = getPlayer(args[1], speaker)[1]
    if not target then notify("No player", args[1] or "") return end
    local stats = {}
    local function scan(container)
        if not container then return end
        for _, v in ipairs(container:GetDescendants()) do
            if v:IsA("IntValue") or v:IsA("NumberValue") or v:IsA("StringValue") or v:IsA("BoolValue") then
                table.insert(stats, v.Name .. " = " .. tostring(v.Value))
            end
        end
    end
    scan(target:FindFirstChild("leaderstats"))
    scan(target:FindFirstChild("Character"))
    for _, v in ipairs(target:GetChildren()) do
        if v:IsA("IntValue") or v:IsA("NumberValue") or v:IsA("StringValue") or v:IsA("BoolValue") then
            table.insert(stats, v.Name .. " = " .. tostring(v.Value))
        end
    end
    notify(target.Name .. " stats", #stats > 0 and table.concat(stats, ", ") or "none")
    for _, s in ipairs(stats) do print("[STAT] " .. s) end
end)
-- Shared recursive stat finder (leaderstats + Character + direct children).
HUB.findStat = function(target, statName)
    if not target or not statName then return nil end
    local found = nil
    local function find(container)
        if found or not container then return end
        for _, v in ipairs(container:GetDescendants()) do
            if v.Name == statName and (v:IsA("IntValue") or v:IsA("NumberValue") or v:IsA("StringValue") or v:IsA("BoolValue")) then
                found = v
                return
            end
        end
    end
    find(target:FindFirstChild("leaderstats"))
    find(target:FindFirstChild("Character"))
    if not found then
        for _, v in ipairs(target:GetChildren()) do
            if v.Name == statName and (v:IsA("IntValue") or v:IsA("NumberValue") or v:IsA("StringValue") or v:IsA("BoolValue")) then
                found = v
                break
            end
        end
    end
    return found
end
addcmd("setstat", {"editstat"}, function(args, speaker)
    local target = getPlayer(args[1], speaker)[1]
    local statName = args[2]
    local value = args[3]
    if not target or not statName or value == nil then notify("Usage: :setstat <player> <stat> <value>", nil) return end
    local stat = HUB.findStat(target, statName)
    if not stat then notify("Stat not found", statName) return end
    if stat:IsA("IntValue") then stat.Value = math.floor(tonumber(value) or 0)
    elseif stat:IsA("NumberValue") then stat.Value = tonumber(value) or 0
    elseif stat:IsA("StringValue") then stat.Value = tostring(value)
    elseif stat:IsA("BoolValue") then stat.Value = (value == "true" or value == "1")
    end
    notify("Set " .. statName, tostring(stat.Value))
end)

            end
        end
    end)

local jerkActive = false

local function startJerk()
    if jerkActive then return end
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildWhichIsA("Humanoid")
    local backpack = LocalPlayer:FindFirstChildWhichIsA("Backpack")
    if not (humanoid and backpack) then
        notify("Jerk", "No character yet")
        return
    end
    jerkActive = true

    local tool = Instance.new("Tool")
    tool.Name = "Jerk Off"
    tool.ToolTip = "in the stripped club. straight up \"jorking it\" . and by \"it\" , haha, well. let's justr say. My peanits."
    tool.RequiresHandle = false
    tool.Parent = backpack
    jerkTool = tool

    local jorkin = false
    local track = nil

    local function stopTomfoolery()
        jorkin = false
        if track then
            track:Stop()
            track = nil
        end
    end

    tool.Equipped:Connect(function() jorkin = true end)
    tool.Unequipped:Connect(stopTomfoolery)
    humanoid.Died:Connect(stopTomfoolery)

    jerkLoop = task.spawn(function()
        while jerkActive and task.wait() do
            if not jorkin then continue end
            local isR15 = humanoid.RigType == Enum.HumanoidRigType.R15
            if not track then
                local anim = Instance.new("Animation")
                anim.AnimationId = not isR15 and "rbxassetid://72042024" or "rbxassetid://698251653"
                track = humanoid:LoadAnimation(anim)
            end
            track:Play()
            track:AdjustSpeed(isR15 and 0.7 or 0.65)
            track.TimePosition = 0.6
            task.wait(0.1)
            while track and track.TimePosition < (not isR15 and 0.65 or 0.7) do task.wait(0.1) end
            if track then
                track:Stop()
                track = nil
            end
        end
    end)
end

    jerkActive = false
    if jerkLoop then
        task.cancel(jerkLoop)
        jerkLoop = nil
    end
    if jerkTool then
        jerkTool:Destroy()
        jerkTool = nil
    end
end

local function FlingTargetPlayer(TargetPlayer)
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter then return end

    local THumanoid, TRootPart, THead, Accessory, Handle
    if TCharacter:FindFirstChildOfClass("Humanoid") then THumanoid = TCharacter:FindFirstChildOfClass("Humanoid") end
    if THumanoid and THumanoid.RootPart then TRootPart = THumanoid.RootPart end
    if TCharacter:FindFirstChild("Head") then THead = TCharacter.Head end
    if TCharacter:FindFirstChildOfClass("Accessory") then Accessory = TCharacter:FindFirstChildOfClass("Accessory") end
    if Accessory and Accessory:FindFirstChild("Handle") then Handle = Accessory.Handle end

    if Character and Humanoid and RootPart then
        if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = RootPart.CFrame end

        if THumanoid and THumanoid.Sit then return FlingNotify("Error", TargetPlayer.Name .. " is sitting") end

        if THead then
            workspace.CurrentCamera.CameraSubject = THead
        elseif Handle then
            workspace.CurrentCamera.CameraSubject = Handle
        elseif THumanoid and TRootPart then
            workspace.CurrentCamera.CameraSubject = THumanoid
        end

        if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

        local FPos = function(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local SFBasePart = function(BasePart)
            local TimeToWait = 2
            local Time = tick()
            local Angle = 0
            repeat
                if RootPart and THumanoid then
                    if BasePart.Velocity.Magnitude < 50 then
                        Angle = Angle + 100
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                        task.wait()
                    end
                end
            until Time + TimeToWait < tick() or not FlingActive
        end

        -- [V6] Save the real fall height before NaN-ing it, so the
        -- fling restores it instead of leaving the workspace broken.
        getgenv().FPDH = workspace.FallenPartsDestroyHeight
        workspace.FallenPartsDestroyHeight = 0 / 0

        local BV = Instance.new("BodyVelocity")
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(0, 0, 0)
        BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)

        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if TRootPart then
            SFBasePart(TRootPart)
        elseif THead then
            SFBasePart(THead)
        elseif Handle then
            SFBasePart(Handle)
        else
            return FlingNotify("Error", TargetPlayer.Name .. " has no valid parts")
        end

        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        workspace.CurrentCamera.CameraSubject = Humanoid

        if getgenv().OldPos then
            repeat
                RootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
                Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
                Humanoid:ChangeState("GettingUp")
                for _, part in pairs(Character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Velocity, part.RotVelocity = Vector3.new(), Vector3.new()
                    end
                end
                task.wait()
            until (RootPart.Position - getgenv().OldPos.p).Magnitude < 25
            workspace.FallenPartsDestroyHeight = getgenv().FPDH
        end
    else
        return FlingNotify("Error", "Your character is not ready")
    end
end

local function FlingStart()
    if FlingActive then return end
    local count = FlingCountSelected()
    if count == 0 then
        if FlingStatusLabel then FlingStatusLabel.Text = "No targets selected!" end
        return
    end
    FlingActive = true
    FlingUpdateStatus()
    FlingLoop = task.spawn(function()
        while FlingActive do
            local validTargets = {}
            for name, player in pairs(FlingTargets) do
                if player and player.Parent then
                    validTargets[name] = player
                else
                    FlingTargets[name] = nil
                    local c = FlingCheckmarks[name]
                    if c then c.Visible = false end
                end
            end
            for _, player in pairs(validTargets) do
                if FlingActive then
                    pcall(FlingTargetPlayer, player)
                    task.wait(0.1)
                else
                    break
                end
            end
            FlingUpdateStatus()
            task.wait(0.5)
        end
    end)
end

local function FlingStop()
    FlingActive = false
    FlingUpdateStatus()
end

end

local function pickMainTarget(player)
    HUB.selectedPlayer = player

    HUB.labels.target.Text = player and ("Target: " .. player.Name) or "Target: none"
    HUB.labels.distance.Text = ""

    HUB.selectedChasePlayer = player

-- V7 auto-target: auto-pick the single enemy-team player for chase (works
-- for killer vs survivor games). Errors (warns) if there are 2+ enemies.
local function autoSelectChaseTarget()
    if not HUB.chaseAutoTarget then return end
    if HUB.selectedChasePlayer and HUB.selectedChasePlayer.Parent then return end
    local myTeam = LocalPlayer.Team
    if not myTeam then return end
    local enemies = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local theirTeam = plr.Team
            if theirTeam and theirTeam ~= myTeam then table.insert(enemies, plr) end
        end
    end
    if #enemies == 1 then
        pickChaseTarget(enemies[1])
        updateChasePlayerList()
        showToast("Auto target: " .. enemies[1].Name, "on")
    elseif #enemies > 1 then
        showToast("Multiple targets (" .. #enemies .. ") - pick manually", "warn")
    end
end

-- ============================================================
-- PLAYER TABLES (V6.1: the classic scrollable inline lists are
-- back -- one on the Main page, one on Chase, replacing the
-- floating picker panel. Rows rebuild on join/leave and on every
-- pick; clicking a row selects, clicking the selected row
-- deselects.)
-- ============================================================
local function updatePlayerList()
    for _, child in ipairs(HUB.playerList:GetChildren()) do
        if child:IsA("TextButton") or child.Name == "EmptyState" then child:Destroy() end
    end
    if HUB.selectedPlayer and not HUB.selectedPlayer.Parent then
        pickMainTarget(nil)
    end
    local n = 0
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            n = n + 1
            local isSel = (player == HUB.selectedPlayer)
            local row = newInstance("TextButton", {
                Size = UDim2.new(1, 0, 0, 24), Text = player.Name,
                TextColor3 = isSel and COLOR.accent or COLOR.text,
                BackgroundColor3 = COLOR.playerBtn,
                Font = currentFontPreset.regular, TextSize = 12, Parent = HUB.playerList,
            })
            round(row, 5)
            bindFont(row, "regular")
            row.MouseButton1Click:Connect(function()
                if isSel then pickMainTarget(nil) else pickMainTarget(player) end
                updatePlayerList()
            end)
            row.MouseEnter:Connect(function()
                if not isSel then row.BackgroundColor3 = COLOR.accentLight end
            end)
            row.MouseLeave:Connect(function()
                if not isSel then row.BackgroundColor3 = COLOR.playerBtn end
            end)
        end
    end
    HUB.playerList.CanvasSize = UDim2.new(0, 0, 0, n * 26 + 4)
    if n == 0 then
        newInstance("TextLabel", {
            Name = "EmptyState",
            Size = UDim2.new(1, 0, 0, 24),
            BackgroundTransparency = 1,
            Text = "No other players",
            TextXAlignment = Enum.TextXAlignment.Center, TextColor3 = COLOR.textDim,
            Font = currentFontPreset.regular, TextSize = 11, Parent = HUB.playerList,
        })
    end
end

local PATH_CLEARANCE = 4.5
local STUCK_CHECK_INTERVAL = 0.75
local STUCK_DISTANCE_THRESHOLD = 2.5
local TARGET_JUMP_THRESHOLD = 8 -- forces a recompute if the watched position moves this far in one frame

local function computePathTo(startPosition, targetPosition)
    local path = PathfindingService:CreatePath({
        AgentRadius = PATH_CLEARANCE, AgentHeight = 5, AgentCanJump = true, AgentCanClimb = true,
    })
    local ok = pcall(function() path:ComputeAsync(startPosition, targetPosition) end)
    if ok and path.Status == Enum.PathStatus.Success then return path:GetWaypoints() end
    return nil
end

-- Raycasts to check if there's a clear line between two points -- used both
-- to skip pathfinding entirely in open areas, and to let an in-progress
-- path "cut the corner" once a wall it was routing around is no longer
-- in the way.
local losRaycastParams = RaycastParams.new()
losRaycastParams.FilterType = Enum.RaycastFilterType.Exclude

local function buildIgnoreList(otherPlayer)
    local list = {}
    if LocalPlayer.Character then table.insert(list, LocalPlayer.Character) end
    if otherPlayer and otherPlayer.Character then table.insert(list, otherPlayer.Character) end
    return list
end

local function hasLineOfSight(fromPos, toPos, ignoreList)
    losRaycastParams.FilterDescendantsInstances = ignoreList
    local result = workspace:Raycast(fromPos, toPos - fromPos, losRaycastParams)
    return result == nil
end

local function newPathState()
    return {
        waypoints = nil, waypointIndex = 1, lastPathTime = 0, lastCommandedIndex = 0,
        currentDestination = nil, stuckCheckPos = nil, stuckCheckTime = 0, stuckStrikes = 0,
        lastSmoothTime = 0, lastWatchPos = nil, lastDirectDestination = nil,
    }
end

-- destinationFn: called only when the path is actually (re)computed
-- watchPositionFn: called every frame -- a big jump here forces an early recompute
-- otherPlayer: used to build the raycast ignore list
local function stepPath(humanoid, myHRP, state, destinationFn, recomputeTime, otherPlayer, watchPositionFn)
    local now = tick()
    local eye = Vector3.new(0, 2, 0)

    local forceRecompute = false
    if watchPositionFn then
        local watchPos = watchPositionFn()
        if state.lastWatchPos and (watchPos - state.lastWatchPos).Magnitude > TARGET_JUMP_THRESHOLD then
            forceRecompute = true
        end
        state.lastWatchPos = watchPos
    end

    if forceRecompute or (not state.waypoints and not state.lastDirectDestination)
        or (state.waypoints and state.waypointIndex > #state.waypoints)
        or (now - state.lastPathTime) > recomputeTime then

        state.lastPathTime = now
        local destination = destinationFn()
        state.currentDestination = destination
        local ignoreList = buildIgnoreList(otherPlayer)

        if hasLineOfSight(myHRP.Position + eye, destination + eye, ignoreList) then
            -- Clear straight shot -- skip pathfinding, walk straight there.
            state.waypoints = nil
            state.lastDirectDestination = destination
            humanoid:MoveTo(destination)
        else
            local points = computePathTo(myHRP.Position, destination)
            if points and #points > 0 then
                state.waypoints, state.waypointIndex = points, math.min(2, #points)
                state.lastCommandedIndex = 0
                state.lastDirectDestination = nil
            else
                state.waypoints = nil
                state.lastDirectDestination = destination
                humanoid:MoveTo(destination)
            end
        end
    elseif state.lastDirectDestination then
        -- Still in a clear line-of-sight window. The humanoid keeps walking to
        -- the last MoveTo point on its own, so no re-issue is needed here; the
        -- recompute above refreshes the destination when the target moves.
    end

    -- Waypoint skipping ("string-pulling"): periodically check if we can see
    -- further ahead than our current waypoint. If so, jump straight to the
    -- furthest visible one instead of walking to each node in turn -- this
    -- is what makes it cut corners around a wall instead of hugging it.
    if state.waypoints and (now - state.lastSmoothTime) > 0.25 then
        state.lastSmoothTime = now
        local ignoreList = buildIgnoreList(otherPlayer)
        for i = #state.waypoints, state.waypointIndex, -1 do
            local wp = state.waypoints[i]
            if hasLineOfSight(myHRP.Position + eye, wp.Position + eye, ignoreList) then
                if i > state.waypointIndex then
                    state.waypointIndex = i
                    state.lastCommandedIndex = 0
                end
                break
            end
        end
    end

    if state.waypoints and state.waypoints[state.waypointIndex] then
        local wp = state.waypoints[state.waypointIndex]
        if state.lastCommandedIndex ~= state.waypointIndex then
            humanoid:MoveTo(wp.Position)
            state.lastCommandedIndex = state.waypointIndex
            if wp.Action == Enum.PathWaypointAction.Jump then humanoid.Jump = true end
        end
        if (myHRP.Position - wp.Position).Magnitude < 4 then
            state.waypointIndex += 1
        end
    end

    -- Stuck detection (unchanged): forces a fresh path/nudge if no ground
    -- is being covered despite an active destination.
    if not state.stuckCheckPos then
        state.stuckCheckPos, state.stuckCheckTime = myHRP.Position, now
    elseif (now - state.stuckCheckTime) > STUCK_CHECK_INTERVAL then
        local moved = (myHRP.Position - state.stuckCheckPos).Magnitude
        if moved < STUCK_DISTANCE_THRESHOLD then
            state.stuckStrikes += 1
            state.waypoints = nil
            state.lastDirectDestination = nil
            state.lastPathTime = 0
            if state.stuckStrikes >= 2 then
                local dest = state.currentDestination or myHRP.Position
                local away = myHRP.Position - dest
                away = (away.Magnitude < 1) and Vector3.new(math.random(-1, 1), 0, math.random(-1, 1)) or away.Unit
                local sideways = Vector3.new(-away.Z, 0, away.X)
                humanoid:MoveTo(myHRP.Position + sideways * 6)
                humanoid.Jump = true
                state.stuckStrikes = 0
            end
        else
            state.stuckStrikes = 0
        end
        state.stuckCheckPos, state.stuckCheckTime = myHRP.Position, now
    end
end

-- ----- Loop manager (exactly one movement loop at a time) -----
local activeMovementLoop = nil
local MovementLoops = {}
local function registerMovementLoop(id, loop)
    MovementLoops[id] = loop
end
local function startMovementLoop(id)
    local loop = MovementLoops[id]
    if not loop then return end
    if activeMovementLoop and activeMovementLoop ~= loop then
        activeMovementLoop.stop()
    end
    loop.start()
    activeMovementLoop = loop
end
local function stopMovementLoop(id)
    local loop = MovementLoops[id]
    if loop then loop.stop() end
    if activeMovementLoop == loop then activeMovementLoop = nil end
end
local function stopAllMovementLoops()
    if activeMovementLoop then activeMovementLoop.stop() end
    activeMovementLoop = nil
end

local chaseOriginalWalkSpeed = 16
HUB.chaseSprintHeld = false

local chaseLoop = {}
chaseLoop.stop = function()
    chaseLoopEnabled = false
    setStatus(HUB.rows.server.chase.status, false)
    HUB.rows.server.chase.set(false, true)
    if chaseConnection then chaseConnection:Disconnect() chaseConnection = nil end
    -- Release sprint/crouch keys + restore the pre-chase WalkSpeed.
    if HUB.chaseSprintHeld then HUB.chaseFireKey(Enum.KeyCode.LeftShift, false) HUB.chaseSprintHeld = false end
    if HUB.chaseCrouchHeld then HUB.chaseFireKey(Enum.KeyCode.LeftControl, false) HUB.chaseCrouchHeld = false end
    if HUB.chaseNoclip then HUB.setChaseNoclip(false) end
    local myHRP, myHumanoid = getParts(LocalPlayer)
    if myHumanoid then myHumanoid.WalkSpeed = chaseOriginalWalkSpeed end
end
chaseLoop.start = function()
    if not HUB.selectedChasePlayer then
        setStatus(HUB.rows.server.chase.status, false, "No target")
        HUB.rows.server.chase.set(false, true)
        showToast("Pick a chase target first", "warn")
        return
    end
    if chaseLoopEnabled then return end
    chaseLoopEnabled = true
    setStatus(HUB.rows.server.chase.status, true)
    local myHRP, myHumanoid = getParts(LocalPlayer)
    if myHumanoid then chaseOriginalWalkSpeed = myHumanoid.WalkSpeed end

    local chaseState = newPathState()
    -- Auto mode: derive smartness live from the situation (target speed,
    -- distance); otherwise use the manual HUB.chaseDifficulty.
    local function aiPresetLimits()
        for _, p in ipairs(HUB.aiPresetOptions) do
            if p.value == HUB.aiPreset then return p end
        end
        return HUB.aiPresetOptions[2]
    end
    local function effDiff()
        if not HUB.chaseAuto then return HUB.chaseDifficulty end
        local tHRP = getParts(HUB.selectedChasePlayer)
        local mHRP = getParts(LocalPlayer)
        if not tHRP or not mHRP then return HUB.chaseDifficulty end
        local tSpeed = tHRP.Velocity and tHRP.Velocity.Magnitude or 0
        local dist = (mHRP.Position - tHRP.Position).Magnitude
        local lim = aiPresetLimits()
        return {
            speed = math.clamp(tSpeed + 8, 14, lim.maxSpeed),
            fleeDistance = math.clamp(25 + tSpeed * 1.5, 15, lim.maxFlee),
            recomputeTime = math.clamp(dist / 40, lim.minReact, 2.0),
            jitter = math.clamp((tSpeed > 20) and 25 or 10, 0, lim.maxJitter),
            triggerMult = 0.55, stopMult = 1.5, minTrigger = 15, minStop = 28,
        }
    end
    -- Barrier radii around the target ("hitbox-aware"): start fleeing when the
    -- target enters the trigger shell, keep fleeing until past the stop shell
    -- (hysteresis stops on/off flicker at the boundary). Difficulty scales the
    -- shell, roughly the 25-50 stud awareness ring.
    -- [FIX 7] Radii derive from the difficulty fields (triggerMult/stopMult/
    -- minTrigger/minStop) instead of the original hardcoded 0.55 / 1.5 / 15 / 28.
    local function currentBarrier()
        local d = effDiff()
        local rad = math.max(d.minTrigger, d.fleeDistance * d.triggerMult)
        -- Always stay outside the orbit/follow radius (keep away from the
        -- target's orbit circle at all costs).
        return rad, math.max(rad * d.stopMult, d.minStop, HUB.followDistance)
    end
    -- Map-aware flee target: fans raycasts across the escape hemisphere (tile
    -- awareness inside the 25-50 stud ring) and picks the direction that is
    -- both away from the threat and clear of walls. If every away path is
    -- blocked (pinned against a wall), it slides along the wall face to
    -- escape -- wall-tamper behavior. The picker re-scores every recompute,
    -- so it never just keeps running the same stale direction.
    local function chasePickFleePoint(fromPos, threatPos)
        local d = effDiff()
        local away = fromPos - threatPos
        if away.Magnitude < 1 then away = Vector3.new(math.random(-10, 10), 0, math.random(-10, 10)) end
        away = away.Unit
        if d.jitter > 0 then
            local a = math.rad(math.random(-d.jitter, d.jitter))
            local ca, sa = math.cos(a), math.sin(a)
            away = Vector3.new(away.X * ca - away.Z * sa, 0, away.X * sa + away.Z * ca)
        end
        local probeRange = math.clamp(d.fleeDistance, 25, 50)
        local ignoreList = buildIgnoreList(HUB.selectedChasePlayer)
        losRaycastParams.FilterDescendantsInstances = ignoreList
        local eye = Vector3.new(0, 2, 0)
        local bestDir, bestScore = nil, -math.huge
        local offsets = { -0.5, -0.35, -0.2, 0, 0.2, 0.35, 0.5 }
        for _, off in ipairs(offsets) do
            local ang = math.rad(off * 70)
            local ca, sa = math.cos(ang), math.sin(ang)
            local dir = Vector3.new(away.X * ca - away.Z * sa, 0, away.X * sa + away.Z * ca).Unit
            local hit = workspace:Raycast(fromPos + eye, dir * probeRange, losRaycastParams)
            local clear = hit and (hit.Position - fromPos).Magnitude or probeRange
            local score = away:Dot(dir) * math.min(clear, probeRange) / probeRange
            if score > bestScore then bestScore, bestDir = score, dir end
        end
        if bestScore < 0.15 then
            -- Pinned against a wall: slide along the wall face with the most room.
            local side = Vector3.new(-away.Z, 0, away.X)
            local h1 = workspace:Raycast(fromPos + eye, side * probeRange, losRaycastParams)
            local h2 = workspace:Raycast(fromPos + eye, -side * probeRange, losRaycastParams)
            local c1 = h1 and (h1.Position - fromPos).Magnitude or probeRange
            local c2 = h2 and (h2.Position - fromPos).Magnitude or probeRange
            bestDir = (c1 >= c2) and side or -side
        end
        return fromPos + bestDir * probeRange
    end

    -- Noclip smartness (Dev/Hacker): cut corners (Dev) or walk through
    -- walls (Hacker) for more distance while fleeing.
    local function chaseShouldNoclip(fleePoint)
        if not HUB.chaseAuto then return false end
        if HUB.aiPreset == "hacker" then return true end
        if HUB.aiPreset == "dev" then
            local myHRP = getParts(LocalPlayer)
            if not myHRP then return false end
            local dir = fleePoint - myHRP.Position
            local dist = dir.Magnitude
            if dist < 1 then return false end
            local params = RaycastParams.new()
            params.FilterDescendantsInstances = { LocalPlayer.Character }
            params.FilterType = Enum.RaycastFilterType.Blacklist
            local hit = workspace:Raycast(myHRP.Position, dir.Unit * dist, params)
            return hit ~= nil
        end
        return false
    end

    chaseConnection = RunService.Heartbeat:Connect(function()
        pcall(function()
            if not HUB.selectedChasePlayer or not HUB.selectedChasePlayer.Parent then
                stopMovementLoop("chase")
                showToast("Chase target left the game", "warn")
                return
            end
            local targetHRP = getParts(HUB.selectedChasePlayer)
            local myHRP, myHumanoid = getParts(LocalPlayer)
            if not targetHRP or not myHumanoid or not myHRP then return end

            local d = effDiff()
            myHumanoid.WalkSpeed = d.speed

            local dist = (myHRP.Position - targetHRP.Position).Magnitude
            local trg, stop = currentBarrier()
            local fleeing = (dist < trg) or (chaseState.mode == "flee" and dist < stop)

            if fleeing then
                chaseState.mode = "flee"
                local fleePoint = chasePickFleePoint(myHRP.Position, targetHRP.Position)
                if chaseShouldNoclip(fleePoint) then
                    HUB.setChaseNoclip(true)
                    myHumanoid:MoveTo(fleePoint)
                else
                    HUB.setChaseNoclip(false)
                    stepPath(
                        myHumanoid, myHRP, chaseState,
                        function() return chasePickFleePoint(myHRP.Position, targetHRP.Position) end,
                        d.recomputeTime, HUB.selectedChasePlayer,
                        function() return targetHRP.Position end
                    )
                end
            else
                -- [FIX 6] Outside the barrier: hold ground instead of running
                -- forever -- one-shot MoveTo to the current position stops the
                -- humanoid dead while staying attached to the ground.
                HUB.setChaseNoclip(false)
                chaseState.mode = "idle"
                chaseState.waypoints = nil
                chaseState.lastDirectDestination = nil
                if myHRP.Velocity.Magnitude > 1 then
                    myHumanoid:MoveTo(myHRP.Position)
                end
            end

            -- Sprint while fleeing, crouch while idle (Shift / Ctrl).
            if HUB.smartSprint and fleeing and not HUB.chaseSprintHeld then
                HUB.chaseFireKey(Enum.KeyCode.LeftShift, true)
                HUB.chaseSprintHeld = true
            elseif (not HUB.smartSprint or not fleeing) and HUB.chaseSprintHeld then
                HUB.chaseFireKey(Enum.KeyCode.LeftShift, false)
                HUB.chaseSprintHeld = false
            end
            if HUB.smartCrouch and not fleeing and not HUB.chaseCrouchHeld then
                HUB.chaseFireKey(Enum.KeyCode.LeftControl, true)
                HUB.chaseCrouchHeld = true
            elseif (not HUB.smartCrouch or fleeing) and HUB.chaseCrouchHeld then
                HUB.chaseFireKey(Enum.KeyCode.LeftControl, false)
                HUB.chaseCrouchHeld = false
            end
        end)
    end)
end

-- ----- Orbit (V7: circle the main target, out-of-reach or free) -----
do
    local orbitOriginalWalkSpeed = 16
    local orbitLoopEnabled = false
    local orbitConnection = nil
    local orbitLoop = {}
    orbitLoop.stop = function()
        orbitLoopEnabled = false
        setStatus(HUB.rows.server.orbit.status, false)
        HUB.rows.server.orbit.set(false, true)
        if orbitConnection then orbitConnection:Disconnect() orbitConnection = nil end
        local myHRP, myHumanoid = getParts(LocalPlayer)
        if myHumanoid then myHumanoid.WalkSpeed = orbitOriginalWalkSpeed end
    end
    orbitLoop.start = function()
        if not HUB.selectedPlayer then
            setStatus(HUB.rows.server.orbit.status, false, "No target")
            HUB.rows.server.orbit.set(false, true)
            showToast("Select a target first", "warn")
            return
        end
        if orbitLoopEnabled then return end
        orbitLoopEnabled = true
        setStatus(HUB.rows.server.orbit.status, true)
        local myHRP, myHumanoid = getParts(LocalPlayer)
        if myHumanoid then orbitOriginalWalkSpeed = myHumanoid.WalkSpeed end

        local orbitState = newPathState()
        local orbitRadius = math.max(HUB.followDistance, 6)
        orbitConnection = RunService.Heartbeat:Connect(function()
            pcall(function()
                if not HUB.selectedPlayer or not HUB.selectedPlayer.Parent then
                    stopMovementLoop("orbit")
                    showToast("Target left the game", "warn")
                    return
                end
                local targetHRP = getParts(HUB.selectedPlayer)
                local myHRP, myHumanoid = getParts(LocalPlayer)
                if not targetHRP or not myHumanoid or not myHRP then return end

                myHumanoid.WalkSpeed = HUB.followSpeed
                local a = tick() * 1.4
                local offset = Vector3.new(math.cos(a) * orbitRadius, 0, math.sin(a) * orbitRadius)

                if HUB.orbitFreedom then
                    -- Freedom: steer around the ring with MoveDirection,
                    -- anchored to the radius and able to go either way.
                    local moveDir = myHumanoid.MoveDirection
                    local toTarget = targetHRP.Position - myHRP.Position
                    local radial = Vector3.new(toTarget.X, 0, toTarget.Z)
                    local rLen = radial.Magnitude
                    radial = (rLen > 0.01) and (radial / rLen) or Vector3.new(1, 0, 0)
                    local tangent = Vector3.new(-radial.Z, 0, radial.X)
                    local dot = moveDir.X * tangent.X + moveDir.Z * tangent.Z
                    if math.abs(dot) < 0.1 then
                        if math.abs(rLen - orbitRadius) > 0.75 then
                            myHumanoid:MoveTo(targetHRP.Position + radial * orbitRadius)
                        end
                    else
                        local ahead = myHRP.Position + ((dot > 0) and tangent or -tangent) * 6
                        local o = Vector3.new(ahead.X, 0, ahead.Z) - Vector3.new(targetHRP.Position.X, 0, targetHRP.Position.Z)
                        local oLen = o.Magnitude
                        o = (oLen > 0.01) and (o * (orbitRadius / oLen)) or (radial * orbitRadius)
                        myHumanoid:MoveTo(targetHRP.Position + o)
                    end
                else
                    -- OOR: snap to the orbit point every frame so the
                    -- target can never reach you.
                    myHRP.CFrame = targetHRP.CFrame + offset
                end
            end)
        end)
    end
    registerMovementLoop("orbit", orbitLoop)
end

showPage = function(page)
    HUB.currentPage = page
    closeActiveOverlay()
    closePickers()
    -- [V6] Reopening CNSL+ keeps the last sub-tab instead of always
    -- bouncing back to CNSL.
    if page == "extras" then showExtrasSubTab(HUB.extrasSubTab) end
    if page == "main" then HUB.showTpSubTab(HUB.tpSubTab) end
    HUB.subTabBar.Visible = (page == "extras")
    HUB.tpSubBar.Visible = (page == "main")
    HUB.pages.main.Visible = (page == "main")
    HUB.pages.chase.Visible = (page == "chase")
    HUB.pages.settings.Visible = (page == "settings")
    HUB.pages.extras.Visible = (page == "extras")
    for _, tab in ipairs({HUB.tabs.main, HUB.tabs.chase, HUB.tabs.settings, HUB.tabs.extras}) do
        local ts = tabStates[tab]
        if ts then ts.refresh() end
    end
    if page == "extras" then
        HUB.frame.Size = UDim2.new(0, HUB.winW, 0, extrasWindowHeight(HUB.extrasSubTab))
    else
        HUB.frame.Size = UDim2.new(0, HUB.winW, 0, HUB.winH)
    end
    if page == "chase" then
        HUB.labels.chaseTarget.Text = "Target: " .. (HUB.selectedChasePlayer and HUB.selectedChasePlayer.Name or "none")
    end
end

-- V7: loop + toggle compatibility -- the re-added Main/Chase page switches
-- mirror the Server sub-tab switches and drive the same loops. The page
-- switch forwards clicks to the server switch (single source of truth);
-- the server switch cb reflects state back so both stay in sync.
do
    local function teleportOn(isOn)
        if isOn then
            if HUB.tpLoop then
                startMovementLoop("teleport")
            else
                doTeleportNow()
                HUB.rows.server.teleport.set(false, true)
            end
        else
            stopMovementLoop("teleport")
        end
    end
    local function chaseOn(isOn)
        if isOn then startMovementLoop("chase") else stopMovementLoop("chase") end
    end

    HUB.rows.main.teleport.cb(function(isOn)
        HUB.rows.server.teleport.set(isOn)
    end)
    HUB.rows.chase.chase.cb(function(isOn)
        HUB.rows.server.chase.set(isOn)
    end)

    HUB.rows.server.teleport.cb(function()
        teleportOn(HUB.rows.server.teleport.get())
        local actual = HUB.rows.server.teleport.get()
        HUB.rows.main.teleport.set(actual, true)
        setStatus(HUB.rows.main.teleport.status, actual)
    end)
    HUB.rows.server.chase.cb(function()
        chaseOn(HUB.rows.server.chase.get())
        local actual = HUB.rows.server.chase.get()
        HUB.rows.chase.chase.set(actual, true)
        setStatus(HUB.rows.chase.chase.status, actual)
    end)

    -- Programmatic state changes (keybinds, loop self-stop) call server
    -- .set(): mirror those onto the page switch + label.
    local teleportSet = HUB.rows.server.teleport.set
    HUB.rows.server.teleport.set = function(v, silent)
        teleportSet(v, silent)
        HUB.rows.main.teleport.set(v, true)
        setStatus(HUB.rows.main.teleport.status, v)
    end
    local chaseSet = HUB.rows.server.chase.set
    HUB.rows.server.chase.set = function(v, silent)
        chaseSet(v, silent)
        HUB.rows.chase.chase.set(v, true)
        setStatus(HUB.rows.chase.chase.status, v)
    end

    -- Loop checkmark (default ON = auto-repeat). Shows a checkmark when
    -- ON (blank when OFF) with an On/Off status label beside it.
    HUB.rows.main.tpLoop.set(true, true)
    setStatus(HUB.rows.main.tpLoop.status, true)
    HUB.rows.main.tpLoop.cb(function(on)
        HUB.tpLoop = on
        setStatus(HUB.rows.main.tpLoop.status, on)
    end)

    -- Stop buttons.
    HUB.buttons.tpStop.MouseButton1Click:Connect(function() stopMovementLoop("teleport") end)
    HUB.buttons.chaseStop.MouseButton1Click:Connect(function() stopMovementLoop("chase") end)
end

-- V7 chase smartness wiring: difficulty is a quick preset; the four
-- knobs override single fields. Auto (Smart AI) locks the manual knobs.
do
    local function optName(options, value)
        for _, o in ipairs(options) do
            if o.value == value then return o.name end
        end
        return tostring(value)
    end
    local function overrideChase(field, value)
        if HUB.chaseAuto then return end
        local merged = {}
        for k, v in pairs(HUB.chaseDifficulty) do merged[k] = v end
        merged[field] = value
        HUB.chaseDifficulty = merged
    end
    local function syncSmartButtons(preset)
        HUB.buttons.fleeDist.Text = "Flee: " .. optName(HUB.fleeDistOptions, preset.fleeDistance)
        HUB.buttons.reaction.Text = "React: " .. optName(HUB.reactionOptions, preset.recomputeTime)
        HUB.buttons.jitter.Text = "Jitter: " .. optName(HUB.jitterOptions, preset.jitter)
        HUB.buttons.walkSpeed.Text = "Walk Speed: " .. optName(SPEED_OPTIONS, preset.speed)
    end
    local function setAutoLock(locked)
        local c = locked and COLOR.textDim or COLOR.text
        HUB.buttons.fleeDist.TextColor3 = c
        HUB.buttons.reaction.TextColor3 = c
        HUB.buttons.jitter.TextColor3 = c
        HUB.buttons.walkSpeed.TextColor3 = c
    end

    createDropdown(HUB.buttons.difficulty, "Difficulty: ", DIFFICULTY_OPTIONS, function(o)
        if HUB.chaseAuto then return end
        HUB.chaseDifficulty = o
        syncSmartButtons(o)
    end)
    createDropdown(HUB.buttons.fleeDist, "Flee: ", HUB.fleeDistOptions, function(o) overrideChase("fleeDistance", o.value) end)
    createDropdown(HUB.buttons.reaction, "React: ", HUB.reactionOptions, function(o) overrideChase("recomputeTime", o.value) end)
    createDropdown(HUB.buttons.jitter, "Jitter: ", HUB.jitterOptions, function(o) overrideChase("jitter", o.value) end)
    createDropdown(HUB.buttons.walkSpeed, "Walk Speed: ", SPEED_OPTIONS, function(o) overrideChase("speed", o.value) end)
    createDropdown(HUB.buttons.aiPreset, "AI Preset: ", HUB.aiPresetOptions, function(o) HUB.aiPreset = o.value end)

    HUB.rows.chase.auto.cb(function(on)
        HUB.chaseAuto = on
        setStatus(HUB.rows.chase.auto.status, on)
        setAutoLock(on)
    end)
    HUB.rows.chase.sprint.cb(function(on)
        HUB.smartSprint = on
        setStatus(HUB.rows.chase.sprint.status, on)
    end)
    HUB.rows.chase.crouch.cb(function(on)
        HUB.smartCrouch = on
        setStatus(HUB.rows.chase.crouch.status, on)
    end)
    HUB.rows.chase.autoTarget.cb(function(on)
        HUB.chaseAutoTarget = on
        setStatus(HUB.rows.chase.autoTarget.status, on)
        if on then autoSelectChaseTarget() end
    end)

    syncSmartButtons(HUB.chaseDifficulty)
end

HUB.buttons.nearest.MouseButton1Click:Connect(function()
    local myHRP = getParts(LocalPlayer)
    if not myHRP then showToast("Can't find your character", "warn") return end

    local closest, closestDist = nil, math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local hrp = getParts(player)
            if hrp then
                local d = (hrp.Position - myHRP.Position).Magnitude
                if d < closestDist then closest, closestDist = player, d end
            end
        end
    end

    if closest then
        pickMainTarget(closest)
    else
        showToast("No other players nearby", "warn")
    end
end)

local minimized = false

local function setMinimized(state)
    minimized = state
    closeActiveOverlay()
    HUB.frame.Size = minimized and MINIMIZED_SIZE or (HUB.currentPage == "extras" and UDim2.new(0, HUB.winW, 0, extrasWindowHeight(HUB.extrasSubTab)) or expandedSize)
    HUB.tabBar.Visible = not minimized

    HUB.tpSubBar.Visible = (not minimized) and HUB.tpSubBar.Visible
    HUB.pages.main.Visible = (not minimized) and HUB.pages.main.Visible
    HUB.pages.chase.Visible = (not minimized) and HUB.pages.chase.Visible
    HUB.pages.settings.Visible = (not minimized) and HUB.pages.settings.Visible
    HUB.pages.extras.Visible = (not minimized) and HUB.pages.extras.Visible
    HUB.minimizeButton.Text = minimized and "+" or "-"
end
HUB.minimizeButton.MouseButton1Click:Connect(function() setMinimized(not minimized) end)

local function togglePopup()
    HUB.frame.Visible = not HUB.frame.Visible
    if not HUB.frame.Visible then
        closeActiveOverlay()
        closePickers()
    end
end

local function closeAndStop()
    stopAllMovementLoops()
    closeActiveOverlay()
    closePickers()
    HUB.frame.Visible = false
end
HUB.closeButton.MouseButton1Click:Connect(closeAndStop)

HUB.frame.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
    local click = input.Position

    if pickerPanel and pickerPanel.Parent then
        local ap, asz = pickerPanel.AbsolutePosition, pickerPanel.AbsoluteSize
        local inPanel = click.X >= ap.X and click.X <= ap.X + asz.X
            and click.Y >= ap.Y and click.Y <= ap.Y + asz.Y
        if not inPanel then closePickers() end
    end

    if activeOverlay then
        local guiPos = click - HUB.screenGui.AbsolutePosition
        local ap, asz = activeOverlay.AbsolutePosition, activeOverlay.AbsoluteSize
        if guiPos.X < ap.X or guiPos.X > ap.X + asz.X or guiPos.Y < ap.Y or guiPos.Y > ap.Y + asz.Y then
            closeActiveOverlay()
        end
    end
end)

-- ============================================================
-- KEYBINDS: T teleport-now, Q follow toggle, C chase toggle, P hub toggle,
-- Tab cycle aim target (when Cycle Mode is on)
-- (Esc intentionally not bound -- it collides with another mapped action)
-- ============================================================
UserInputService.InputBegan:Connect(function(input, gameProcessedEvent)
    if gameProcessedEvent then return end
    local key = input.KeyCode

    if key == Enum.KeyCode.T then
        doTeleportNow()
    elseif key == Enum.KeyCode.Tab then
        HUB.cycleAimTarget()
    elseif key == Enum.KeyCode.Q then
        HUB.rows.server.follow.set(not HUB.rows.server.follow.get())
    elseif key == Enum.KeyCode.C then
        HUB.rows.server.chase.set(not HUB.rows.server.chase.get())
    elseif key == Enum.KeyCode.P then
        togglePopup()
    end
end)
