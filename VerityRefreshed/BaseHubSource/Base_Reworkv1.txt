-- =====================================================================
-- BASE :: SCHIZHUB BASE REWORK v1
-- A from-scratch rebuild of the v7 hub. Data-driven engine, one
-- component factory, modular feature registry, an event bus, and a
-- live companion face (NOVA) that reacts to everything you do.
--
-- DESIGN RULES
--   1. EVERYTHING IS DATA. Presets, options, features, face states are
--      plain tables; behaviour is derived from them, never copy-pasted.
--   2. ONE FACTORY. UI is built by UI.* builders that return handles.
--      Styling flows from THEME + CONFIG, so restyling is global.
--   3. FEATURES ARE MODULES. Each feature registers itself once (rows,
--      commands, keybinds, run/stop); the registry renders it.
--   4. EVENTS DECOUPLE. Systems publish to EVENTS; the face and the UI
--      subscribe. Adding a reaction is one line, not a callback spiral.
--   5. POSITIONS ARE DATA. Pages are built with a flow cursor + named
--      slots, so reordering a page is editing a table, not editing px.
--   6. STATE IS OBSERVABLE. Loops expose run/stop + status; the toggle
--      watchdog is a generic guard, not a per-feature special case.
-- =====================================================================

-- PART 2. UTIL -- small, pure helpers

local UTIL = {}

function UTIL.round(n) return math.floor(n + 0.5) end

function UTIL.clamp(n, lo, hi) return math.max(lo, math.min(hi, n)) end

function UTIL.lerp(a, b, t) return a + (b - a) * t end

function UTIL.mixColor(a, b, t)
    return Color3.fromRGB(
        UTIL.round(UTIL.lerp(a.R * 255, b.R * 255, t)),
        UTIL.round(UTIL.lerp(a.G * 255, b.G * 255, t)),
        UTIL.round(UTIL.lerp(a.B * 255, b.B * 255, t))
    )
end

-- Brighten or darken a colour by an amount in -1..1.
function UTIL.shade(c, amt)
    local f = 1 + amt
    return Color3.fromRGB(
        UTIL.clamp(UTIL.round(c.R * 255 * f), 0, 255),
        UTIL.clamp(UTIL.round(c.G * 255 * f), 0, 255),
        UTIL.clamp(UTIL.round(c.B * 255 * f), 0, 255)
    )
end

-- Split a string on a delimiter. `plain` treats the delimiter as a
-- literal string (no pattern magic), which is what command chaining
-- needs for "\" and other magic characters.
function UTIL.split(s, delim, plain)
    if type(s) ~= "string" then return {} end
    if delim == "" or delim == nil then return { s } end
    local out = {}
    local dl = #delim
    local i = 1
    while true do
        local j = s:find(delim, i, plain)
        if not j then
            out[#out + 1] = s:sub(i)
            break
        end
        out[#out + 1] = s:sub(i, j - 1)
        i = j + dl
    end
    return out
end

-- Cheap stable hash for tagging generated names.
function UTIL.hash(s)
    local h = 5381
    for i = 1, #s do
        h = (h * 33 + s:byte(i)) % 0xFFFFFFFF
    end
    return h
end

-- Nearest player by a provided distance function; returns the best or nil.
function UTIL.nearest(players, distFn)
    local best, bestD = nil, math.huge
    for _, p in ipairs(players) do
        local d = distFn(p)
        if d < bestD then best, bestD = p, d end
    end
    return best
end

-- A tiny but sufficient pub/sub bus. Events are fire-and-forget;
-- handlers run synchronously in fire order.
local EVENTS = { _handlers = {} }
function EVENTS.on(event, fn)
    EVENTS._handlers[event] = EVENTS._handlers[event] or {}
    table.insert(EVENTS._handlers[event], fn)
    return function()
        local list = EVENTS._handlers[event]
        if list then
            for i, h in ipairs(list) do
                if h == fn then table.remove(list, i) break end
            end
        end
    end
end
function EVENTS.fire(event, ...)
    local list = EVENTS._handlers[event]
    if list then
        local copy = table.clone(list)
        for _, fn in ipairs(copy) do
            pcall(fn, ...)
        end
    end
end

local function edgingFilterColor(filter)
    if not filter then return COLOR.border end
    if filter.mode == "bg" then return COLOR.bg end
    if filter.mode == "accent" then return COLOR.accent end
    if filter.mode == "accent2" then return COLOR.accent2 end
    return COLOR.border
end

        bg = bg,

        accent2 = acc2,

        accentDark = UTIL.shade(acc, -0.35),
        accent2Light = UTIL.shade(acc2, 0.35),
        accent2Dark = UTIL.shade(acc2, -0.35),

local themedNodes = {}
local refreshTheme -- forward-declared: applyTheme runs before its definition
local hubUIScale -- created by PART 7 (shell); declared here so appliers can use it
local function applyTheme(preset)
    currentThemePreset = preset
    for k, v in pairs(buildColorTable(preset)) do
        COLOR[k] = v
    end
    refreshTheme()
end

local themedFonts = {}
local function bindFont(instance, weight)
    table.insert(themedFonts, { instance = instance, base = instance.TextSize or 12 })
    instance.Font = (weight == "bold" or weight == "regular") and (weight == "bold" and currentFontPreset.bold or currentFontPreset.regular) or currentFontPreset.regular
end

local function applyTextScale(factor)
    for _, e in ipairs(themedFonts) do
        if e.instance and e.instance.Parent then
            e.instance.TextSize = math.max(CONFIG.textMin, math.floor((e.base or 12) * factor))
        end
    end
end

local function applySize(preset)
    currentSizePreset = preset
    local scale = TweenService:Create(hubUIScale, TweenInfo.new(0.15), { Scale = preset.scale })
    scale:Play()
    applyTextScale((preset.text or 1) * (currentTextPreset.factor or 1))
end

local function applyFont(preset)
    currentFontPreset = preset
    for _, e in ipairs(themedFonts) do
        if e.instance and e.instance.Parent then
            e.instance.Font = preset.regular
        end
    end
end

local function applyTextPreset(preset)
    currentTextPreset = preset
    applyTextScale((currentSizePreset.text or 1) * preset.factor)
end

local function applyBorderStyle(preset)
    currentBorderPreset = preset
    if HUB and HUB.frame then
        HUB.frame.BorderSizePixel = preset.thickness
        local corner = HUB.frame:FindFirstChildOfClass("UICorner")
        if corner then corner.CornerRadius = UDim.new(0, preset.radius) end
    end
end

local function applyEdgingFilter(filter)
    currentEdgingFilter = filter
    COLOR.border = edgingFilterColor(filter)
    refreshTheme()
    EVENTS.fire("edgingChanged", filter)
end

-- Recolour every live UI node that opted into theming. Nodes register
-- themselves under UI.themed[] with {obj, key(s), color-ref}; recolour is a
-- pure sweep, so any node added later can be re-painted at any time.
refreshTheme = function()
    for _, node in ipairs(themedNodes) do
        local obj = node.obj
        if obj and obj.Parent then
            for _, spec in ipairs(node.specs) do
                if spec.role then
                    local c = COLOR[spec.role]
                    if c then obj[spec.prop or "BackgroundColor3"] = c end
                elseif spec.color then
                    obj[spec.prop or "BackgroundColor3"] = spec.color
                end
            end
        end
    end
end

-- =====================================================================
-- PART 5. UI -- the single component factory
-- Every visual is built through here. Builders return a handle object
-- (or a plain list) so callers never reach into instances. Styling
-- always flows from COLOR / CONFIG; adding a themed node is one call.

local UI = {}

local function newInstance(className, props)
    local obj = Instance.new(className)
    if props then
        for k, v in pairs(props) do obj[k] = v end
    end
    return obj
end

-- Register a node for live recolour. specs: { {role="accent", prop="BackgroundColor3"}, ... }
function UI.theme(obj, specs)
    table.insert(themedNodes, { obj = obj, specs = specs })
    return obj
end

function UI.round(obj, radius)
    local corner = newInstance("UICorner", { CornerRadius = UDim.new(0, radius or 8) })
    corner.Parent = obj
    return corner
end

function UI.stroke(obj, color, thickness)
    local edge = newInstance("UIStroke", { Color = color or COLOR.border, Thickness = thickness or 1 })
    edge.Parent = obj
    return edge
end

function UI.frame(opts)
    local f = newInstance("Frame", {
        Name = opts.name or "Frame",
        Size = opts.size or UDim2.new(0, 100, 0, 20),
        Position = opts.position or UDim2.new(0, 0, 0, 0),
        BackgroundColor3 = opts.color or COLOR.panel,
        BackgroundTransparency = opts.transparency or 0,
        BorderSizePixel = opts.border or 0,
        ZIndex = opts.z or 1,
        Visible = opts.visible == nil and true or opts.visible,
        Parent = opts.parent,
    })
    if opts.radius then UI.round(f, opts.radius) end
    if opts.stroke then UI.stroke(f, opts.stroke, opts.strokeThickness or 1) end
    return f
end

function UI.label(opts)
    local l = newInstance("TextLabel", {
        Name = opts.name or "Label",
        Size = opts.size or UDim2.new(0, 100, 0, 18),
        Position = opts.position or UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 1,
        Text = opts.text or "",
        TextColor3 = opts.color or COLOR.text,
        Font = currentFontPreset.regular,
        TextSize = opts.textSize or 12,
        TextXAlignment = opts.x or Enum.TextXAlignment.Center,
        TextYAlignment = opts.y or Enum.TextYAlignment.Center,
        TextWrapped = opts.wrap or false,
        ZIndex = opts.z or 1,
        Visible = opts.visible == nil and true or opts.visible,
        Parent = opts.parent,
    })
    bindFont(l, opts.weight)
    if opts.colorRole then UI.theme(l, { { role = opts.colorRole, prop = "TextColor3" } }) end
    return l
end

function UI.button(opts)
    local b = newInstance("TextButton", {
        Name = opts.name or "Button",
        Size = opts.size or UDim2.new(0, 120, 0, 26),
        Position = opts.position or UDim2.new(0, 0, 0, 0),
        BackgroundColor3 = opts.color or COLOR.panel,
        BackgroundTransparency = opts.transparency or 0,
        Text = opts.text or "",
        TextColor3 = opts.textColor or COLOR.text,
        Font = currentFontPreset.bold,
        TextSize = opts.textSize or 12,
        AutoButtonColor = false,
        ZIndex = opts.z or 2,
        Visible = opts.visible == nil and true or opts.visible,
        Parent = opts.parent,
    })
    bindFont(b, "bold")
    if opts.radius then UI.round(b, opts.radius) end
    if opts.stroke then UI.stroke(b, opts.stroke, opts.strokeThickness or 1) end
    if opts.colorRole then UI.theme(b, { { role = opts.colorRole, prop = "BackgroundColor3" } }) end
    if opts.onClick then b.MouseButton1Click:Connect(function() pcall(opts.onClick) end) end
    if opts.hover ~= false then
        local base = b.BackgroundColor3
        local glow = opts.hoverColor or UTIL.shade(base, 0.18)
        b.MouseEnter:Connect(function()
            if b.Visible then
                b.BackgroundColor3 = glow
                if opts.hoverText then b.TextColor3 = opts.hoverText end
            end
        end)
        b.MouseLeave:Connect(function()
            if b.Visible then
                b.BackgroundColor3 = base
                if opts.hoverText then b.TextColor3 = opts.textColor or COLOR.text end
            end
        end)
    end
    return b
end

-- A labelled toggle row. Returns { row, statusLabel, set, get, onChange }.
-- The "set" function is a guarded setter: it updates state, the label,
-- and fires EVENTS.toggle (which the face + watchdog listen to).
function UI.toggleRow(opts)
    local row = UI.frame({
        name = opts.name or "ToggleRow",
        size = UDim2.new(1, -10, 0, opts.height or CONFIG.rowH),
        position = UDim2.new(0, 5, 0, opts.y or 0),
        color = COLOR.bg,
        transparency = 0,
        parent = opts.parent,
        z = opts.z or 2,
    })
    local label = UI.label({
        text = opts.text or "Toggle",
        size = UDim2.new(1, -86, 1, 0),
        position = UDim2.new(0, 10, 0, 0),
        x = Enum.TextXAlignment.Left,
        color = COLOR.text,
        textSize = opts.textSize or 12,
        weight = opts.weight or "regular",
        parent = row,
        colorRole = nil,
    })
    -- A real switch: track + sliding knob. State is obvious at a glance.
    local track = UI.frame({
        name = "Switch",
        size = UDim2.new(0, 34, 0, 14),
        position = UDim2.new(1, -46, 0.5, -7),
        color = COLOR.switchOff,
        radius = 7,
        parent = row,
        z = 3,
    })
    local knob = UI.frame({
        name = "Knob",
        size = UDim2.new(0, 12, 0, 12),
        position = UDim2.new(0, 1, 0.5, -6),
        color = COLOR.text,
        radius = 6,
        parent = track,
        z = 4,
    })
    local state = opts.default or false
    local cb = opts.onChange
    local function set(v, silent)
        v = v == nil and (not state) or v
        state = v
        track.BackgroundColor3 = state and COLOR.on or COLOR.switchOff
        knob.Position = UDim2.new(state and 1 or 0, state and -13 or 1, 0.5, -6)
        row.BackgroundColor3 = state and UTIL.shade(COLOR.accent, -0.65) or COLOR.bg
        if cb and not silent then pcall(cb, state) end
        EVENTS.fire("toggle", { id = opts.id, on = state, label = opts.text })
    end
    local btn = UI.button({
        text = "",
        size = UDim2.new(1, 0, 1, 0),
        position = UDim2.new(0, 0, 0, 0),
        color = Color3.new(1, 1, 1),
        transparency = 1,
        parent = row,
        onClick = function() set() end,
    })
    if opts.id and HUB.toggleSets then
        HUB.toggleSets[opts.id] = { set = set, get = function() return state end }
    end

function UI.slider(opts)
    local row = UI.frame({

        parent = row,
    })
    local fill = UI.frame({
        size = UDim2.new(0, 0, 1, 0),
        position = UDim2.new(0, 0, 0, 0),
        color = COLOR.accent,
        radius = 2,
        parent = track,
    })
    UI.theme(fill, { { role = "accent", prop = "BackgroundColor3" } })
    local knob = UI.frame({
        size = UDim2.new(0, 14, 0, 14),
        position = UDim2.new(0, -7, 0.5, -7),
        color = COLOR.text,
        radius = 7,
        parent = row,
        z = 3,
    })
    local value = opts.default or opts.min or 0
    local min, max = opts.min or 0, opts.max or 100
    local onChange = opts.onChange
    local function set(v, silent)
        value = UTIL.clamp(v, min, max)
        local t = (value - min) / (max - min)
        fill.Size = UDim2.new(t, 0, 1, 0)
        knob.Position = UDim2.new(t, -7, 0.5, -7)
        if onChange and not silent then pcall(onChange, value) end
    end
    -- Drag support (uses only Frame-valid events)
    local dragging = false
    local function updateFromX(x)
        local abs = row.AbsolutePosition
        local w = row.AbsoluteSize.X
        set(min + (x - abs.X) / math.max(w, 1) * (max - min))
    end
    row.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            updateFromX(input.Position.X)
        end
    end)
    row.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            updateFromX(input.Position.X)
        end
    end)
    if opts.absoluteX then set(value, true) end
    return row, set, function() return value end
end

-- =====================================================================
-- PART 5b. COLLAPSIBLE SECTIONS
-- The RED 101 contract is preserved exactly: every section snapshots its
-- content ONCE from the final un-shifted layout (HUB.finalizeCollapsibles
-- at end of script). items/off/contentH are constants after that, so any
-- collapse/expand order can never spill content into a neighbour.
-- =====================================================================
local function nextSectionEntry(parent, entry)
    local best, bestCur = nil, math.huge
    for _, e in ipairs(HUB.sectionHeaders) do
        if e.parent == parent and e.curY and e.curY > entry.curY and e.curY < bestCur then
            best, bestCur = e, e.curY
        end
    end
    return best
end

function UI.section(opts)
    local parent = opts.parent
    local y = opts.y
    local shiftParent = opts.shiftParent or parent
    local row = UI.frame({
        name = opts.name or ("Section_" .. opts.text),
        size = UDim2.new(1, -10, 0, CONFIG.headerH),
        position = UDim2.new(0, 5, 0, y),
        color = COLOR.bg,
        parent = parent,
        z = opts.z or 2,
    })
    local tick = UI.frame({
        size = UDim2.new(0, 3, 0, 12),
        position = UDim2.new(0, 0, 0.5, -6),
        color = COLOR.accent2,
        radius = 1.5,
        parent = row,
        z = 3,
    })
    UI.theme(tick, { { role = "accent2", prop = "BackgroundColor3" } })
    local label = UI.label({
        text = opts.text or "",
        size = UDim2.new(1, -24, 1, 0),
        position = UDim2.new(0, 10, 0, 0),
        x = Enum.TextXAlignment.Left,
        color = COLOR.textDim,
        textSize = opts.textSize or 11,
        weight = "bold",
        parent = row,
    })
    local chevron = UI.label({
        text = "▾",
        size = UDim2.new(0, 12, 1, 0),
        position = UDim2.new(1, -16, 0, 0),
        color = COLOR.textDim,
        textSize = 10,
        weight = "bold",
        parent = row,
    })
    local click = UI.button({
        text = "",
        size = UDim2.new(1, 0, 1, 0),
        position = UDim2.new(0, 0, 0, 0),
        color = Color3.new(1, 1, 1),
        transparency = 1,
        parent = row,
        hover = false,
    })
    local entry = {
        parent = parent, y = y, curY = y,
        collapsed = false, contentH = nil, items = nil, finalized = false,
    }
    local function finalizeSection()
        if entry.finalized then return end
        entry.finalized = true
        local nextE = nextSectionEntry(parent, entry)
        local bottom = nextE and nextE.curY or (HUB.contentBottom[parent] or (entry.curY + 20))
        entry.contentH = math.max(0, bottom - entry.curY - CONFIG.headerH)
        entry.items = {}
        for _, child in ipairs(parent:GetChildren()) do
            if child ~= row then
                local py = child.Position and child.Position.Y
                local cy = py and py.Offset
                if cy and cy > entry.curY and cy < bottom then
                    entry.items[#entry.items + 1] = { obj = child, off = cy - entry.curY }
                end
            end
        end
    end
    entry.finalize = finalizeSection
    local function shiftBelow(delta, fromY)
        if delta == 0 then return end
        fromY = fromY or (entry.curY + CONFIG.headerH)
        for _, child in ipairs(shiftParent:GetChildren()) do
            if child ~= row and child.Visible ~= false and child.Position then
                local py = child.Position.Y
                local cy = py and py.Offset
                if cy and cy >= fromY then
                    child.Position = UDim2.new(child.Position.X.Scale, child.Position.X.Offset, child.Position.Y.Scale, cy + delta)
                end
            end
        end
        for _, e in ipairs(HUB.sectionHeaders) do
            if (e.parent == shiftParent or e.parent == parent) and e.curY and e.curY >= fromY then
                e.curY = e.curY + delta
            end
        end
    end
    local function toggle()
        entry.collapsed = not entry.collapsed
        label.TextColor3 = entry.collapsed and COLOR.off or COLOR.textDim
        tick.Visible = not entry.collapsed
        chevron.Text = entry.collapsed and "▸" or "▾"
        chevron.TextColor3 = entry.collapsed and COLOR.off or COLOR.textDim
        local nextE = nextSectionEntry(parent, entry)
        local nextCurY = nextE and nextE.curY or (HUB.contentBottom[parent] or (entry.curY + 20))
        finalizeSection()
        local contentH = entry.contentH
        local items = entry.items
        if entry.collapsed then
            for _, it in ipairs(items) do
                pcall(function() it.obj.Visible = false end)
            end
            shiftBelow(-contentH, nextCurY)
        else
            shiftBelow(contentH, nextCurY)
            for _, it in ipairs(items) do
                pcall(function()
                    it.obj.Position = UDim2.new(it.obj.Position.X.Scale, it.obj.Position.X.Offset, it.obj.Position.Y.Scale, entry.curY + it.off)
                    it.obj.Visible = true
                end)
            end
        end
        if HUB.refreshCanvas then HUB.refreshCanvas(shiftParent) end
        if HUB.fitExtrasWindow then HUB.fitExtrasWindow(shiftParent) end
        EVENTS.fire("section", { id = opts.id, text = opts.text, collapsed = entry.collapsed })
    end
    entry.toggle = toggle
    click.MouseButton1Click:Connect(toggle)
    click.MouseEnter:Connect(function()
        label.TextColor3 = COLOR.text
        chevron.TextColor3 = COLOR.text
    end)
    click.MouseLeave:Connect(function()
        if not entry.collapsed then
            label.TextColor3 = COLOR.textDim
            chevron.TextColor3 = COLOR.textDim
        end
    end)
    table.insert(HUB.sectionHeaders, entry)
    return row, label
end

local function ensureToastPool(n)
    while #toastPool < n do
        local t = UI.frame({
            name = "Toast",
            size = UDim2.new(0, 0, 0, CONFIG.toastH),
            position = UDim2.new(0, 0, 0, 0),
            color = COLOR.titleBar,
            radius = 8,
            stroke = COLOR.border,
            strokeThickness = 1,
            parent = HUB.notifyLayer,
            visible = false,
            z = 60,
        })
        UI.label({
            text = "",
            size = UDim2.new(1, -12, 1, 0),
            position = UDim2.new(0, 6, 0, 0),
            x = Enum.TextXAlignment.Left,
            color = COLOR.text,
            textSize = 12,
            parent = t,
        })
        table.insert(toastPool, t)
    end
    while #toastPool > n do
        local t = table.remove(toastPool)
        t:Destroy()
    end
end

local function showToast(text, tone)
    if HUB.useDrawing then
        pcall(function()
            local cam = workspace.CurrentCamera
            local toast = Drawing.new("Text")
            toast.Text = tostring(text)
            toast.Size = 18
            toast.Color = (tone=="on" or tone=="success") and Color3.fromRGB(80,255,120) or (tone=="warn" and Color3.fromRGB(255,200,80) or Color3.fromRGB(230,230,230))
            toast.Position = V2New((cam and cam.ViewportSize.X/2 or 400), 38)
            toast.Center = true
            toast.Outline = true
            toast.Visible = true
            task.delay(CONFIG.toastLifetime, function() pcall(function() toast:Remove() end) end)
        end)
        pcall(function() print("[TOAST] "..tostring(text)) end)
        pcall(function() EVENTS.fire("toast", { text=text, tone=tone }) end)
        return
    end
    ensureToastPool(CONFIG.toastMax)
    -- move older toasts up
    for _, t in ipairs(toastPool) do
        if t.Visible then
            t.Position = UDim2.new(1, -t.AbsoluteSize.X - 14, 0, t.Position.Y.Offset - CONFIG.toastH - 6)
        end
    end
    local t = nil
    for _, cand in ipairs(toastPool) do
        if not cand.Visible then t = cand break end
    end
    t = t or toastPool[#toastPool]
    if not t then return end
    local label = t:FindFirstChildOfClass("TextLabel")
    if not label then return end
    label.Text = text
    local col = COLOR.textDim
    if tone == "on" or tone == "success" then col = COLOR.on
    elseif tone == "warn" or tone == "error" then col = COLOR.warn
    elseif tone == "accent" then col = COLOR.accent
    end
    label.TextColor3 = col
    t.BackgroundColor3 = col == COLOR.warn and UTIL.shade(COLOR.warn, -0.7) or COLOR.titleBar
    t.Position = UDim2.new(1, -t.AbsoluteSize.X - 14, 0, t.Position.Y.Offset + 0)
    t.Visible = true
    t.ZIndex = 60
    task.delay(CONFIG.toastLifetime, function() toastClose(t) end)
    EVENTS.fire("toast", { text = text, tone = tone })
end

-- PART 6. VERITY -- the companion / 2D mascot
-- Hybrid build: hub stays ScreenGui, Verity portrait is a modular 2D
-- Frame stack (Drawing-like layers). Hair off by default, toggleable.
-- Layers: Head -> Hair -> Eyes -> Eyebrows -> Mouth -> Effects
-- States: neutral, focused, curious, amused, annoyed, alert, glitch
-- Reference: yellow smiley (clean circle, intense eyes, simple smile,
-- dark muted accents, crisp outlines, flat cel, glitch = per-layer shift)
-- =====================================================================
local VERITY = {}
local FACE = VERITY -- alias so old code (HUB.face, FACE.set) keeps working
HUB.verity = VERITY

VERITY.opts = table.clone(VERITY_DEFAULTS or FACE_DEFAULTS)
VERITY.state = "neutral"
VERITY.lastEvent = { event = "", time = 0 }
VERITY.hairOn = false

local verityRoot, verityHead, verityHair, verityGlitchR, verityGlitchB
local eyeL, eyeR, pupilL, pupilR, browL, browR, mouth

local function tween(inst, props, time)
    time = time or CONFIG.tweenDefault
    local t = TweenService:Create(inst, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local SHAPES = {
    round = { corner = 100, w = 9, h = 10 },
    narrow = { corner = 100, w = 6, h = 11 },
    wide = { corner = 100, w = 11, h = 7 },
    sharp = { corner = 2, w = 7, h = 11 },
}

local function eyeGeometry(shapeKey, size)
    local s = SHAPES[shapeKey] or SHAPES.round
    return { w = s.w * (size or 1), h = s.h * (size or 1), corner = s.corner }
end

local function pupilSize(style, fw)
    local s = fw / 34
    if style == "dot" then return 3 * s, 3 * s
    elseif style == "slit" then return 1.5 * s, 6 * s
    elseif style == "star" then return 5 * s, 5 * s
    end
    return 2 * s, 3 * s
end

-- Verity palette: yellow head, black features, accent from theme
local VERITY_HEAD = Color3.fromRGB(255, 220, 55)
local VERITY_HEAD_DARK = Color3.fromRGB(235, 200, 40)
local VERITY_FEATURE = Color3.fromRGB(16, 16, 16)
local VERITY_ACCENT = Color3.fromRGB(90, 90, 255)

function VERITY.apply()
    if not verityHead then return end
    local o = VERITY.opts
    local fw, fh = CONFIG.faceW, CONFIG.faceH
    local g = eyeGeometry(o.shape or "round", o.size)
    local eyeW, eyeH = g.w, g.h
    local sep = fw * 0.28
    eyeL.Size = UDim2.new(0, eyeW, 0, eyeH)
    eyeR.Size = UDim2.new(0, eyeW, 0, eyeH)
    local corner = UDim.new(0, g.corner)
    if eyeL:FindFirstChildOfClass("UICorner") then eyeL:FindFirstChildOfClass("UICorner").CornerRadius = corner end
    if eyeR:FindFirstChildOfClass("UICorner") then eyeR:FindFirstChildOfClass("UICorner").CornerRadius = corner end
    eyeL.Position = UDim2.new(0.5, -(sep + eyeW/2), 0.5, -eyeH/2 - 2)
    eyeR.Position = UDim2.new(0.5, -eyeW/2 + sep, 0.5, -eyeH/2 - 2)
    eyeL.BackgroundColor3 = o.color or VERITY_FEATURE
    eyeR.BackgroundColor3 = o.color or VERITY_FEATURE
    -- hair visibility follows VERITY.hairOn and opts.hair (off by default)
    if verityHair then
        local wantsHair = VERITY.hairOn and (o.hair ~= false)
        verityHair.Visible = wantsHair and true or false
    end
    VERITY.setMouth(VERITY.state)
end

local MOUTH_GLYPH = {
    neutral = "–", focused = "–", curious = "o", amused = "⌣", annoyed = "—", alert = "o", glitch = "⌣",
    idle="–", happy="⌣", tired="–", sad="⌢", shocked="o", sleep="z", line="–", smile="⌣", frown="⌢",
}

function VERITY.setMouth(state)
    if not mouth then return end
    local g = MOUTH_GLYPH[state] or MOUTH_GLYPH.neutral
    mouth.Text = g
    -- mouth color stays black for smiley; tint for glitch
    if state == "glitch" then
        mouth.TextColor3 = VERITY_ACCENT
    else
        mouth.TextColor3 = VERITY_FEATURE
    end
end

local function browForState(state, on)
    if state == "focused" then return -3, -3
    elseif state == "annoyed" then return -4, -4
    elseif state == "curious" then return 3, -1
    elseif state == "alert" then return 2, 2
    elseif state == "amused" then return -1, -2
    end
    return 0, 0
end

local blinking = false
local function doBlink()
    if blinking or not verityHead then return end
    if VERITY.state == "sleep" then return end
    blinking = true
    local dur = CONFIG.faceBlinkDuration
    local style = VERITY.opts.blinkStyle
    if style == "quick" then dur = 0.06 elseif style == "slow" then dur = 0.28 elseif style == "heavy" then dur = 0.4 end
    local g = eyeGeometry(VERITY.opts.shape or "round", VERITY.opts.size)
    tween(eyeL, { Size = UDim2.new(0, g.w, 0, 1) }, dur*0.4)
    tween(eyeR, { Size = UDim2.new(0, g.w, 0, 1) }, dur*0.4)
    task.delay(dur*0.4+0.02, function()
        if not verityHead then return end
        tween(eyeL, { Size = UDim2.new(0, g.w, 0, g.h) }, dur*0.6)
        tween(eyeR, { Size = UDim2.new(0, g.w, 0, g.h) }, dur*0.6)
        task.delay(dur*0.6, function() blinking = false end)
    end)
end

local function scheduleBlink()
    if not VERITY.opts.reactsTo then return end
    local min, max = CONFIG.faceBlinkMin, CONFIG.faceBlinkMax
    task.delay(min + math.random()*(max-min), function()
        if verityHead and verityHead.Visible then doBlink(); scheduleBlink() end
    end)
end

RunService.RenderStepped:Connect(function()

            if pupilR then pupilR.Position = UDim2.new(0.5, ox, 0.5, oy) end
        end
    end
end)

function VERITY.setHair(on)

function VERITY.set(state, reason, opts)
    local o = opts or {}
    -- allow new Verity states plus legacy FACE states
    local mapLegacy = { idle="neutral", happy="amused", tired="neutral", sad="annoyed", shocked="alert", sleep="neutral", focus="focused", party="amused" }
    if mapLegacy[state] then state = mapLegacy[state] end
    if VERITY.state == state and not o.force then return end
    VERITY.state = state
    -- mouth + brows
    VERITY.setMouth(state)
    local ly, ry = browForState(state, true)
    if browL then browL.Position = UDim2.new(0, 1, 0, ly) end
    if browR then browR.Position = UDim2.new(1, -11, 0, ry) end
    -- eye shape per state
    local shape = "round"
    if state=="focused" then shape="narrow" elseif state=="alert" then shape="round" elseif state=="amused" then shape="wide" end
    local g = eyeGeometry(shape, VERITY.opts.size)
    if eyeL then tween(eyeL, { Size = UDim2.new(0, g.w, 0, g.h) }, 0.08) end
    if eyeR then tween(eyeR, { Size = UDim2.new(0, g.w, 0, g.h) }, 0.08) end
    -- head tilt for curious
    if verityHead then
        local rot = 0
        if state=="curious" then rot = 5 elseif state=="amused" then rot = -3 end
        verityHead.Rotation = rot
    end
    -- glitch layers only for corrupted
    local isGlitch = state=="glitch"
    if verityGlitchR then verityGlitchR.Visible = isGlitch end
    if verityGlitchB then verityGlitchB.Visible = isGlitch end
    if isGlitch and verityHead then
        -- random per-layer jitter next frame
        task.delay(0.06, function()
            if VERITY.state~="glitch" then return end
            if verityGlitchR then verityGlitchR.Position = UDim2.new(0, math.random(-2,2), 0, math.random(-1,1)) end

        end)
    end

        tween(verityHead, { Size = UDim2.new(0, CONFIG.faceW*1.04, 0, CONFIG.faceH*1.04) }, CONFIG.tweenSnappy)

        end)
    end

end

-- =====================================================================
-- PART 8. PAGES -- tab pages + extras sub-views
-- Pages are ScrollingFrames (except Extras which hosts its own sub-views):
-- content taller than the window scrolls via CanvasSize, placed through a
-- sequential flow cursor (HUB.pageSections maps section -> feature ids).
-- Headers always sit directly above their rows, never over a neighbour.
-- =====================================================================
local function pageCanvas(parent, startY)
    local y = startY or CONFIG.flowPad
    return {
        frame = parent,
        header = function(text, id)
            y = y + CONFIG.flowPad
            local row, _ = UI.section({ text = text, y = y, parent = parent, id = id })
            y = y + CONFIG.headerH + CONFIG.headerPad
            return row
        end,
        rowY = function()
            local v = y
            y = y + CONFIG.rowH + CONFIG.rowGap
            return v
        end,
        add = function(h)
            y = y + (h or 0) + CONFIG.sectionGap
        end,
        pos = function()
            return y
        end,
        reset = function(v) y = v end,
    }
end

local function pageFrame(id, scroll)
    local p
    if scroll ~= false then
        p = Instance.new("ScrollingFrame")
        p.Name = id
        p.Size = UDim2.new(1, 0, 1, -(CONFIG.titleH + CONFIG.tabH))
        p.Position = UDim2.new(0, 0, 0, CONFIG.titleH + CONFIG.tabH)
        p.BackgroundColor3 = COLOR.bg
        p.BackgroundTransparency = 0
        p.BorderSizePixel = 0
        p.CanvasSize = UDim2.new(0, 0, 0, 800)
        p.ScrollBarThickness = 6
        pcall(function() p.ScrollingDirection = Enum.ScrollingDirection.Y end)
        p.Active = true
        p.ClipsDescendants = true
        p.Visible = false
        p.Parent = HUB.frame
    else
        p = Instance.new("Frame")
        p.Name = id
        p.Size = UDim2.new(1, 0, 1, -(CONFIG.titleH + CONFIG.tabH))
        p.Position = UDim2.new(0, 0, 0, CONFIG.titleH + CONFIG.tabH)
        p.BackgroundColor3 = COLOR.bg
        p.BackgroundTransparency = 0
        p.BorderSizePixel = 0
        p.Visible = false
        p.Parent = HUB.frame
    end
    return p
end

-- PART 11. ENGINE -- loops, movement, teleport, chase, aim, commands
-- Every loop goes through runLoop/stopLoop so stopEverything() and the
-- UI watchdog are one generic mechanism.
-- =====================================================================
local engine = {}
HUB.engine = engine
_G.engine = engine

-- ---- loop manager -----------------------------------------------------
local loops = {}
local function runLoop(id, fn)
    if loops[id] then return end
    loops[id] = RunService.Heartbeat:Connect(function(dt)
        pcall(fn, dt)
    end)
end
local function stopLoop(id)
    local c = loops[id]
    if c then
        pcall(c.Disconnect, c)
        loops[id] = nil
    end
end
local function anyLoops()
    return next(loops) ~= nil
end

HUB.stopEverything = function()
    for id in pairs(loops) do stopLoop(id) end
    if LocalPlayer and LocalPlayer.Character then
        local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then hum.WalkSpeed = 16 hum.JumpPower = 50 end
    end
    if _G.OLDCF then pcall(function() workspace.CurrentCamera.CFrame = _G.OLDCF end) end
    _G.OLDCF = nil
    if _G.OLDFOV then pcall(function() workspace.CurrentCamera.FieldOfView = _G.OLDFOV end) end
    _G.OLDFOV = nil
    showToast("Stopped all loops", "warn")
end

-- ---- helpers ----------------------------------------------------------
local function char()
    return LocalPlayer and LocalPlayer.Character
end
local function hrp()
    local c = char()
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function hum()
    local c = char()
    return c and c:FindFirstChildOfClass("Humanoid")
end

    set = function(v)
        if v then
            runLoop("noclip", function()
                local root = hrp()
                local p = root and root.Position
                if root and p then
                    for _, part in ipairs(char():GetChildren()) do
                        if part:IsA("BasePart") and part.CanCollide then
                            part.CanCollide = false
                        end
                    end
                end
            end)
        else
            stopLoop("noclip")
        end
        showToast(v and "Noclip on" or "Noclip off", v and "on" or "textDim")
    end,
}

-- =====================================================================
-- FLY
-- =====================================================================
engine.fly = {
    speed = 20,
    setSpeed = function(v)
        engine.fly.speed = v
        showToast("Fly speed: " .. tostring(v), "accent")
    end,
    set = function(v)
        if v then
            runLoop("fly", function(dt)
                dt = dt or 0.05
                local root = hrp()
                local h = hum()
                if not root or not h then return end
                local cam = workspace.CurrentCamera
                local cf = cam and cam.CFrame or CFrame.new(0, 5, 0)
                local forward = (cf.LookVector * Vector3.new(1, 0, 1)).Unit
                local right = (cf.RightVector * Vector3.new(1, 0, 1)).Unit
                local speed = engine.fly.speed
                local keys = UserInputService:GetKeyboardState()
                local vel = Vector3.new(0, 0, 0)
                if keys[Enum.KeyCode.W] then vel = vel + forward * speed end
                if keys[Enum.KeyCode.S] then vel = vel - forward * speed end
                if keys[Enum.KeyCode.A] then vel = vel - right * speed end
                if keys[Enum.KeyCode.D] then vel = vel + right * speed end
                if keys[Enum.KeyCode.Space] then vel = vel + Vector3.new(0, speed * 0.8, 0) end
                if keys[Enum.KeyCode.LeftShift] then vel = vel - Vector3.new(0, speed * 0.8, 0) end
                root.Velocity = Vector3.new(vel.X, vel.Y * 0.4, vel.Z)
                root.CFrame = root.CFrame + vel * dt
            end)
        else
            stopLoop("fly")
            local root = hrp()

engine.chase = {
    on = false,
    mode = "Nearest",
    difficulty = DIFFICULTY_OPTIONS[1],
    flee = 35,
    speed = 20,
    setTargetMode = function(m) engine.chase.mode = m end,
    setDifficulty = function(opt) engine.chase.difficulty = opt end,
    setFlee = function(v) engine.chase.flee = v end,
    setSpeed = function(v) engine.chase.speed = v end,
    set = function(v)
        engine.chase.on = v
        if v then
            runLoop("chase", function(dt)
                local root = hrp()
                if not root then return end
                local t = engine.chase.pickTarget()
                local tChar = t and (t.Character or workspace:FindFirstChild(t.Name))
                local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
                if not tRoot then return end
                local dist = (tRoot.Position - root.Position).Magnitude
                local d = engine.chase.difficulty
                if dist < d.fleeDistance then
                    local dir = (root.Position - tRoot.Position).Unit * Vector3.new(1, 0, 1)
                    root.CFrame = CFrame.new(root.Position + (dir * d.speed * dt))
                    EVENTS.fire("chase", { state = "flee", dist = dist })
                else
                    local dir = (tRoot.Position - root.Position).Unit * Vector3.new(1, 0, 1)
                    root.CFrame = CFrame.new(root.Position + (dir * engine.chase.speed * dt))
                    EVENTS.fire("chase", { state = "chase", dist = dist })
                end
            end)
        else
            stopLoop("chase")
        end
        EVENTS.fire("chase", { state = v and "start" or "stop" })
        EVENTS.fire("toggle", { id = "chase-enable", on = v, label = "Chase on" })
        if HUB.toggleSets and HUB.toggleSets["chase-enable"] then
            pcall(HUB.toggleSets["chase-enable"].set, v, true)
        end
        showToast(v and "Chase on" or "Chase off", v and "on" or "textDim")
    end,
    pickTarget = function()
        local my = hrp()
        if not my then return nil end
        local best, bestD = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local c = p.Character
                local r = c and c:FindFirstChild("HumanoidRootPart")
                if r then
                    local d = (r.Position - my.Position).Magnitude
                    if d < bestD then best, bestD = p, d end
                end
            end
        end
        return best
    end,
}

local function aimTick()
    if not engine.aim.on then return end
    if engine.aim.keyMode and not UserInputService:IsKeyDown(Enum.KeyCode.F) then return end
    local cam = workspace.CurrentCamera
    local myRoot = hrp()
    if not cam or not myRoot then return end
    local best, bestA = nil, engine.aim.fov
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local c = p.Character
            local part = c and c:FindFirstChild(engine.aim.part) or (c and c:FindFirstChild("Head"))
            if part then
                local sp = cam:WorldToScreenPoint(part.Position)
                local mousePos = UserInputService:GetMouseLocation()
                local dx, dy = sp.X - mousePos.X, sp.Y - mousePos.Y
                local ang = math.sqrt(dx * dx + dy * dy)
                if ang < bestA then best, bestA = part, ang end
            end
        end
    end
    if best then
        cam.CFrame = CFrame.lookAt(cam.CFrame.Position, best.Position)
    end
end

local commandList = {}
-- Map command names to { featureId, toggleState } so addcmd can sync
-- the UI toggle after the engine fires.
local cmdToggleMap = {}

function addcmd(name, args)
    name = tostring(name):lower()
    if commandList[name] then
        local ok, err = pcall(commandList[name].fn, args)
        if not ok then
            showToast("Command error: " .. tostring(err), "error")
            EVENTS.fire("error", err)
        end
        -- Sync the UI toggle (if one exists) so pill/keybind clicks
        -- update the visual switch, not just the engine state.
        local map = cmdToggleMap[name]
        if map and HUB.toggleSets and HUB.toggleSets[map.id] then
            HUB.toggleSets[map.id].set(map.state, true)
        end
    else
        showToast("Unknown command: " .. name ..  " (try /help)", "warn")
    end
end

local function cmd(name, fn, desc, featureId, featureState)
    commandList[name] = { fn = fn, desc = desc or name }
    if featureId and featureState ~= nil then
        cmdToggleMap[name] = { id = featureId, state = featureState }
    end
end

-- =====================================================================
-- execCmd -- the terminal / chain engine
-- chains split on "\", each segment trimmed then dispatched
-- =====================================================================
function execCmd(text)
    if type(text) ~= "string" or text == "" then return end
    local parts = UTIL.split(text, "\\", true)
    for _, part in ipairs(parts) do
        local p = part:gsub("^%s+", ""):gsub("%s+$", "")
        if p ~= "" then
            local name, args = p:match("^(%S+)%s*(.*)$")
            name = (name or p):lower()
            args = args or ""
            if name == "tppos" then
                addcmd(name, args)
            else
                addcmd(name, args)
            end
            EVENTS.fire("command", { name = name, args = args })
        end
    end
end

-- Client-side boot diagnostics: run once immediately and again ~0.7s in.
local function diag()
    local function absInfo(obj)
        local p, s = obj.AbsolutePosition, obj.AbsoluteSize
        return (p and ("%d,%d"):format(p.X, p.Y) or "nil"), (s and ("%d,%d"):format(s.X, s.Y) or "nil")
    end
    local ap, as = absInfo(HUB.frame)
    print(("[DIAG] frame visible=%s abs=%s size=%s"):format(tostring(HUB.frame.Visible), ap, as))
    for _, pid in ipairs({ "main", "chase", "settings", "extras" }) do
        local page = HUB.pages[pid]
        if page then
            local n = 0
            for _ in pairs(page:GetChildren()) do n = n + 1 end
            local paps, pass = absInfo(page)
            local extra = ""
            if page:IsA("ScrollingFrame") then
                local cv = page.CanvasSize
                local cp = page.CanvasPosition
                extra = (" canvasY=%s posY=%s"):format(cv and tostring(cv.Y.Offset) or "-", cp and tostring(cp.Y) or "-")
            end
            print(("[DIAG] page %s visible=%s children=%d abs=%s size=%s%s"):format(
                pid, tostring(page.Visible), n, paps, pass, extra))
        end
    end
end
diag()

    diag()
end)
